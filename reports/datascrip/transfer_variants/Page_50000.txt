    LOCAL PROCEDURE GetLastWhsShptNo@1();
    VAR
      WhsShptHeader@1000 : Record 7320;
      TrnfShptHeader@1001 : Record 5744;
    BEGIN
      //T20.01
      IF "No." <> '' THEN
      BEGIN
        WhsShptHeader.RESET;
        WhsShptHeader.SETRANGE(WhsShptHeader."Line Source Document", WhsShptHeader."Line Source Document"::"Outbound Transfer");
        WhsShptHeader.SETRANGE(WhsShptHeader."Line Source No.", "No.");
        IF WhsShptHeader.FINDLAST THEN
          WhsShptNo := WhsShptHeader."No."
        ELSE
          WhsShptNo := '';

        IF WhsShptNo = '' THEN
        BEGIN
          TrnfShptHeader.RESET;
          TrnfShptHeader.SETCURRENTKEY(TrnfShptHeader."Transfer Order No.");
          TrnfShptHeader.SETRANGE(TrnfShptHeader."Transfer Order No.", "No.");
          IF TrnfShptHeader.FINDLAST THEN
            WhsShptNo := TrnfShptHeader."Shipment No.";
        END;
      END
      ELSE
        WhsShptNo := '';
      //End T20.01
    END;

    END;
