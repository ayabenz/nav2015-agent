    VAR
      SalesHeader@1002 : Record 36;
      SalesPost@1001 : Codeunit 80;

      Text001@1004 : TextConst 'ENU=Do you want to post the %1?';
      Text002@1003 : TextConst 'ENU=&Invoice';

    LOCAL PROCEDURE Code@1();
    BEGIN
      // D34.03
      WITH SalesHeader DO BEGIN
      // YE:050902
        TESTFIELD(Status,Status::Released);
      // YE

        CASE "Document Type" OF
          "Document Type"::Order:
            BEGIN
             Ship := FALSE;   // BNY 20.11.26
             Invoice := TRUE;
            END;
          "Document Type"::"Return Order":
            BEGIN
              Receive := FALSE;
              Invoice := TRUE;
            END ELSE
              IF NOT
                 CONFIRM(
                   Text001,FALSE,
                   "Document Type")
              THEN
                EXIT;
        END;
        SalesPost.RUN(SalesHeader);
      END;
      // D34.03
    END;
