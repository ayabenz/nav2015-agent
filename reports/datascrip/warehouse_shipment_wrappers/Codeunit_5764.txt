OBJECT Codeunit 5764 Whse.-Post Ship. (Ship/Invoic)
{
  OBJECT-PROPERTIES
  {
    Date=12/10/15;
    Time=01:00:00;
    Modified=Yes;
    Version List=NAVW17.00,DTS:T14;
  }
  PROPERTIES
  {
    TableNo=7321;
    OnRun=BEGIN
            CheckPickIsFinished(Rec); //T14.01
            WhseShipLine.COPY(Rec);
            Code;
            Rec := WhseShipLine;
          END;

  }
  CODE
  {
    VAR
      WhseShipLine@1001 : Record 7321;
      WhsePostShipment@1002 : Codeunit 5763;
      Selection@1003 : Integer;
      Text000@1006 : TextConst 'ENU=&Ship,Ship &and Invoice';
      Text50000@1000 : TextConst 'ENU=All Registered Pick Lines must be finished before posting.';

    PROCEDURE Code@1();
    VAR
      Invoice@1001 : Boolean;
    BEGIN
      WITH WhseShipLine DO BEGIN
        IF FIND THEN BEGIN
          Selection := STRMENU(Text000,1);
          IF Selection = 0 THEN
            EXIT;
          Invoice := (Selection = 2);
        END;

        WhsePostShipment.SetPostingSettings(Invoice);
        WhsePostShipment.SetPrint(FALSE);
        WhsePostShipment.RUN(WhseShipLine);
        WhsePostShipment.GetResultMessage;
        CLEAR(WhsePostShipment);
      END;
    END;

    PROCEDURE CheckPickIsFinished@2(par_WhseShipmLine@1003 : Record 7321);
    VAR
      lRec_Loc@1000 : Record 14;
      lRec_Pick@1001 : Record 5772;
      lRec_WhseShipmHeader@1002 : Record 7320;
    BEGIN
      //T14.01
      lRec_WhseShipmHeader.RESET;
      IF lRec_WhseShipmHeader.GET(par_WhseShipmLine."No.") THEN BEGIN
        lRec_Loc.RESET;
        lRec_Loc.SETRANGE(Code,lRec_WhseShipmHeader."Location Code");
        lRec_Loc.SETRANGE("Require Pick",TRUE);
        IF lRec_Loc.FINDFIRST THEN BEGIN
          lRec_Pick.RESET;
          lRec_Pick.SETRANGE(Type,lRec_Pick.Type::Pick);
          lRec_Pick.SETRANGE("Shipment/Receipt No.",lRec_WhseShipmHeader."No.");
          IF lRec_Pick.FINDLAST THEN BEGIN
            IF lRec_Pick.Finished = FALSE THEN
              ERROR(Text50000);
          END;
        END;
      END;
      //T14.01
    END;

    BEGIN
    {
      T14.01    RT    10 Dec 2015   Add function CheckPickIsRegistered
    }
    END.
  }
}
