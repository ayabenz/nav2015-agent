    PROCEDURE GetRemainingAmount@1000000027(EntryNo@1000000000 : Integer) : Decimal;
    VAR
      LDetVendLedgEntry@1000000001 : Record 380;
      TotalRemainingAmount@1000000002 : Decimal;
    BEGIN
      TotalRemainingAmount := 0;
      LDetVendLedgEntry.RESET;
      LDetVendLedgEntry.SETCURRENTKEY("Vendor Ledger Entry No.","Posting Date");
      LDetVendLedgEntry.SETRANGE("Vendor Ledger Entry No.",EntryNo);
      REPEAT
         TotalRemainingAmount += LDetVendLedgEntry.Amount;
      UNTIL LDetVendLedgEntry.NEXT = 0;
      EXIT(TotalRemainingAmount);
    END;

    PROCEDURE SetLimitRemAmt@1000000001(PLimitRemAmt@1000000000 : Decimal;PLimitRemAmt2@1000000005 : Decimal);
    VAR
      LRemAmtPercentage@1000000001 : Decimal;
      LDetVendLedgEntry@1000000002 : Record 380;
      LAmount@1000000003 : Decimal;
      LRemAmount@1000000004 : Decimal;
    BEGIN
      IF TypeFilter > 0 THEN
      BEGIN
         //MARKEDONLY(FALSE);
         RESET;
         IF FINDSET THEN
         REPEAT
            LAmount := 0;
            LRemAmount := 0;
            LDetVendLedgEntry.RESET;
            LDetVendLedgEntry.SETCURRENTKEY("Vendor Ledger Entry No.","Entry Type","Posting Date");
            LDetVendLedgEntry.SETRANGE("Vendor Ledger Entry No.","Entry No.");
            IF LDetVendLedgEntry.FIND('-') THEN
            BEGIN
               REPEAT
                  IF LDetVendLedgEntry."Entry Type" IN [
                  LDetVendLedgEntry."Entry Type"::"Initial Entry",
                  LDetVendLedgEntry."Entry Type"::"Unrealized Loss",
                  LDetVendLedgEntry."Entry Type"::"Unrealized Gain",
                  LDetVendLedgEntry."Entry Type"::"Realized Loss",
                  LDetVendLedgEntry."Entry Type"::"Realized Gain",
                  LDetVendLedgEntry."Entry Type"::"Payment Discount",
                  LDetVendLedgEntry."Entry Type"::"Payment Discount (VAT Excl.)",
                  LDetVendLedgEntry."Entry Type"::"Payment Discount (VAT Adjustment)",
                  LDetVendLedgEntry."Entry Type"::"Payment Tolerance",
                  LDetVendLedgEntry."Entry Type"::"Payment Discount Tolerance",
                  LDetVendLedgEntry."Entry Type"::"Payment Tolerance (VAT Excl.)",
                  LDetVendLedgEntry."Entry Type"::"Payment Tolerance (VAT Adjustment)",
                  LDetVendLedgEntry."Entry Type"::"Payment Discount Tolerance (VAT Excl.)",
                  LDetVendLedgEntry."Entry Type"::"Payment Discount Tolerance (VAT Adjustment)"] THEN
                     LAmount += LDetVendLedgEntry.Amount
                  ELSE IF (LDetVendLedgEntry."Entry Type"=LDetVendLedgEntry."Entry Type"::Application) AND
                  (LDetVendLedgEntry."Document Type"=LDetVendLedgEntry."Document Type"::"Credit Memo")THEN
                     LAmount += LDetVendLedgEntry.Amount;   //tambahan hendra 30 Maret 2010
                  LRemAmount += LDetVendLedgEntry.Amount;
               UNTIL LDetVendLedgEntry.NEXT = 0;
               IF (LAmount <> 0) THEN
                 LRemAmtPercentage := 100*LRemAmount/LAmount
               ELSE
                 LRemAmtPercentage := 0
            END;

               MARK(TRUE)
            ELSE
            IF (TypeFilter = 2) AND (LRemAmtPercentage = PLimitRemAmt) THEN
               MARK(TRUE)
            ELSE
            IF (TypeFilter = 3) AND (PLimitRemAmt <= LRemAmtPercentage) AND (LRemAmtPercentage <= PLimitRemAmt2) THEN
               MARK(TRUE)
            ELSE
               MARK(FALSE);
         UNTIL NEXT = 0;
         MARKEDONLY(TRUE);
         CurrPage.UPDATE(FALSE);
      END;
    END;
