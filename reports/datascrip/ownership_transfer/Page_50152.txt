    PROCEDURE UpdateBalance@1000000013();
    VAR
      tmpOT@1000000000 : Record 50037;
    BEGIN
      TotalBalance := 0;

      tmpOT.COPY(Rec);
      IF tmpOT.FIND('-') THEN
      REPEAT
        TotalBalance += tmpOT.Quantity;
      UNTIL tmpOT.NEXT = 0;

      Balance := 0;

      tmpOT.SETRANGE("Document No.", Rec."Document No.");
      tmpOT.SETRANGE("Item No.", Rec."Item No.");
      IF tmpOT.FIND('-') THEN
      REPEAT
        Balance += tmpOT.Quantity;
      UNTIL tmpOT.NEXT = 0;
    END;

    PROCEDURE CheckBalance@1000000000();
    VAR
      tOT@1000000000 : Record 50037;
      tOT2@1000000001 : Record 50037;
    BEGIN
      tOT.COPY(Rec);
      IF tOT.FIND('-') THEN
      REPEAT
        tOT2.RESET;
        tOT2.SETRANGE("Item No.", tOT."Item No.");
        TotalBalance := 0;
        IF tOT2.FIND('-') THEN
        REPEAT
           TotalBalance += tOT2.Quantity;
        UNTIL tOT2.NEXT = 0;

        IF TotalBalance > 0 THEN
           ERROR('Transfer Owner tidak balance untuk Item '+ tOT."Item No.");

      UNTIL tOT.NEXT = 0;
    END;

    PROCEDURE SetDetail@1000000001();

        gBol_DescEditable := TRUE;
        gBol_LocationEditable := TRUE;
        gBol_OwnerEditable := TRUE;
        gBol_QuantityEditable := TRUE;
      END;
    END;

    END.
