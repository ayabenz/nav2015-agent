    PROCEDURE AssistEdit@1000000001(OldP2BHeader@1000000000 : Record 50013) : Boolean;
    VAR
      Rec_P2BHeader@1000000001 : Record 50013;
    BEGIN
      WITH P2BHeader DO BEGIN
        COPY(Rec);
        GetInventorySetup;
        TestNoSeries;
        IF NoSeriesMgt.SelectSeries(GetNoSeriesCode,OldP2BHeader."No. Series","No. Series") THEN BEGIN
          NoSeriesMgt.SetSeries("No.");
          IF Rec_P2BHeader.GET("No.") THEN
            ERROR(Text50000,"No.");
          Rec := P2BHeader;
          EXIT(TRUE);
        END;
      END;
    END;

    PROCEDURE GetNoSeriesRelationship@1000000004() : Code[10];
    VAR
      NoSeriesRelationship@1000000000 : Record 310;
