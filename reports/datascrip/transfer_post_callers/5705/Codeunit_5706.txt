  CODE
  {
    VAR
      Text000@1000 : TextConst 'ENU=&Ship,&Receive';
      TransHeader@1001 : Record 5740;
      Text001@1004 : TextConst 'ENU=Appl.-to Item Entry must be filled';
      Text002@1003 : TextConst 'ENU=Salesperson code must be filled';
      Text003@1002 : TextConst 'ENU=Sorry can''t process, Please check outstanding loan first';
      ItemJournalLine@1005 : Record 83;

    LOCAL PROCEDURE Code@1();
    VAR
      TransLine@1000 : Record 5741;
      DefaultNumber@1001 : Integer;
      TransferPostShipment@1002 : Codeunit 5704;
      TransferPostReceipt@1003 : Codeunit 5705;
      Selection@1004 : ' ,Shipment,Receipt';
      LocationRec@1009 : Record 14;
      ItemRec@1008 : Record 27;
      TrnfShptHdr@1007 : Record 5744;
      ItemJnlLine@1006 : Record 83;
      P2BLine@1005 : Record 50014;
      tTL@1011 : Record 5741;
      tItem@1010 : Record 27;
      Rec_ReservEntry@1012 : Record 337;
      Bol_NeedCheck@1013 : Boolean;
    BEGIN
      WITH TransHeader DO BEGIN

      IF "Journal Batch Name" <> '' THEN
        BEGIN
          ItemJournalLine.RESET;
          ItemJournalLine.SETFILTER("Journal Template Name", TransHeader."Journal Template Name");
          ItemJournalLine.SETFILTER("Journal Batch Name", TransHeader."Journal Batch Name");
          IF ItemJournalLine.FINDFIRST THEN
           ERROR('Item Journal Line No. '+ItemJournalLine."Document No."+' belum diposting.');
        END;

         { IF ( "Transfer-from Code" = 'SR') OR ("Transfer-to Code" = 'SR') THEN
          BEGIN
             ERROR('Lokasi SR sedang dalam proses stocktake');
          END;}

       {
       IF "No." <> '' THEN BEGIN
          tTL.RESET;
          tTL.SETRANGE("Document No.", "No.");
          IF tTL.FINDFIRST THEN REPEAT
            tItem.RESET;
            tItem.SETRANGE("No.",tTL."Item No.");
            tItem.SETRANGE("Global Dimension 1 Code",'102');
            IF tItem.FINDFIRST THEN
               ERROR('Item ' + FORMAT(tTL."Item No.")
                         + ' sedang tidak dapat digunakan, '
                         + 'karena sedang dalam proses stocktake');

            {
             IF tItem.GET(tTL."Item No.") THEN
              IF tItem."Item Category Code" IN
                ['104-70000'] THEN
                    ERROR('Item ' + FORMAT(tTL."Item No.")
                        + ' sedang tidak dapat digunakan, '
                        + 'karena sedang dalam proses stoktake');
            }
          UNTIL tTL.NEXT = 0;
        END;
        }
        //T20.1
        IF ("Transfer Type" = "Transfer Type"::Transfer) OR ("Transfer Type" = "Transfer Type"::Loan) THEN
        BEGIN
          TESTFIELD("Transfer-from Code");
          LocationRec.GET("Transfer-from Code");
          IF LocationRec.Ownership THEN
            IF "From Location Ownership" = '' THEN
            BEGIN
              TransLine.RESET;
              TransLine.SETRANGE(TransLine."Document No.", "No.");
              TransLine.SETFILTER(TransLine.Quantity, '>%1', 0);
              IF TransLine.FIND('-') THEN
              REPEAT
                TransLine.TESTFIELD(TransLine.Owner);
              UNTIL TransLine.NEXT = 0;
            END
            ELSE
              TESTFIELD("From Location Ownership");
        END;

        IF ("Transfer Type" = "Transfer Type"::Transfer) OR ("Transfer Type" = "Transfer Type"::"Return Loan") THEN
        BEGIN
          TESTFIELD("Transfer-to Code");
          LocationRec.GET("Transfer-to Code");
          IF LocationRec.Ownership THEN
            IF "To Location Ownership" = '' THEN
            BEGIN
              TransLine.RESET;
              TransLine.SETRANGE(TransLine."Document No.", "No.");
              TransLine.SETFILTER(TransLine.Quantity, '>%1', 0);
              IF TransLine.FIND('-') THEN
              REPEAT
                TransLine.TESTFIELD(TransLine.Owner);
              UNTIL TransLine.NEXT = 0;
            END
            ELSE
              TESTFIELD("To Location Ownership");
        END;
        //End T20.1

        TransLine.SETRANGE("Document No.","No.");
        IF TransLine.FIND('-') THEN
          REPEAT
            IF (TransLine."Quantity Shipped" < TransLine.Quantity) AND
               (DefaultNumber = 0)
            THEN
              DefaultNumber := 1;
            IF (TransLine."Quantity Received" < TransLine.Quantity) AND
               (DefaultNumber = 0)
            THEN
              DefaultNumber := 2;
          UNTIL (TransLine.NEXT = 0) OR (DefaultNumber > 0);

        IF DefaultNumber = 0 THEN
          DefaultNumber := 1;
        Selection := STRMENU(Text000,DefaultNumber);
        CASE Selection OF
          0:
            EXIT;
          //T20.3
          {
          1:
            TransferPostShipment.RUN(TransHeader);
          2:
            TransferPostReceipt.RUN(TransHeader);
          }
          1:
          BEGIN
            AddLotNo;

            IF TransHeader."Transfer Type" = TransHeader."Transfer Type"::"Return Loan" THEN
            BEGIN
              TransLine.SETRANGE("Document No.","No.");
              TransLine.SETFILTER(Quantity,'<>%1',0);
              IF TransLine.FIND('-') THEN
              REPEAT
                IF ItemRec.GET(TransLine."Item No.") AND ((ItemRec."Item Tracking Code" = '') OR (ItemRec."Item Tracking Code" <> '')) THEN BEGIN
                  //T00.01
                  Bol_NeedCheck := TRUE;
                  Rec_ReservEntry.RESET;
                  Rec_ReservEntry.SETCURRENTKEY("Source ID","Source Ref. No.","Source Type","Source Subtype","Source Batch Name","Source Prod. Order Line","Reservation Status","Shipment Date","Expected Receipt Date");
                  Rec_ReservEntry.SETRANGE("Source ID", TransLine."Document No.");
                  Rec_ReservEntry.SETRANGE("Source Ref. No.", TransLine."Line No.");
                  Rec_ReservEntry.SETRANGE("Source Type", 5741);
                  Rec_ReservEntry.SETRANGE("Source Subtype", 0);
                  Rec_ReservEntry.SETRANGE("Item No.", TransLine."Item No.");
                  IF Rec_ReservEntry.FINDFIRST THEN
                    Bol_NeedCheck := FALSE;
                  //T00.01
                  IF (TransLine."Appl.-to Item Entry" = 0) AND (TransLine."Qty. to Ship" > 0) AND Bol_NeedCheck THEN
                    ERROR(Text001);
                END;
              UNTIL (TransLine.NEXT = 0);
            END;

            IF (TransHeader."Transfer Type" = TransHeader."Transfer Type"::Loan) THEN
            BEGIN
              IF (TransHeader."Salesperson Code" = '') THEN
                ERROR(Text002);
            END ;

            TransferPostShipment.RUN(TransHeader);
          END;

          2:
          BEGIN
            IF "Transfer Type" IN ["Transfer Type"::Transfer,"Transfer Type"::Loan] THEN
            BEGIN
              TrnfShptHdr.RESET;
              TrnfShptHdr.SETRANGE(TrnfShptHdr."Transfer Order No.", TransHeader."No.");
              IF TrnfShptHdr.FIND('-') THEN
              REPEAT
                IF LocationRec.GET(TrnfShptHdr."Transfer-from Code") THEN
                  IF LocationRec.Ownership THEN
                    TrnfShptHdr.TESTFIELD("Truck Status", TrnfShptHdr."Truck Status"::Jalan);
              UNTIL TrnfShptHdr.NEXT = 0;
            END;

            //skip checking, changed to automatic posting
            {
            IF (TransHeader."P2B Document No."<>'') THEN BEGIN
              ItemJnlLine.RESET;
              ItemJnlLine.SETFILTER("P2B Document No.",'<>%1','');
              ItemJnlLine.SETRANGE("Journal Template Name", TransHeader."Journal Template Name");
              ItemJnlLine.SETRANGE("Journal Batch Name", TransHeader."Journal Batch Name");
              IF ItemJnlLine.FINDFIRST THEN BEGIN
                 P2BLine.RESET;
                 P2BLine.SETRANGE("Document No.", ItemJnlLine."P2B Document No.");
                 IF P2BLine.FINDFIRST THEN BEGIN
                    IF P2BLine.Status = P2BLine.Status::Close THEN
                      ERROR('%1 belum terposting..', ItemJnlLine."Document No.")
                    ELSE IF P2BLine.Status = P2BLine.Status::Release THEN
                       ERROR('%1 gagal Posting, Item Journal No. %2 harap dihapus terlebih dahulu..',ItemJnlLine."P2B Document No.",ItemJnlLine."Document No.")
                    ELSE
                       ERROR('Hubungin MIS untuk pengecekan Error...');
                 END
              END;
            END;
            }

            IF (TransHeader."Transfer Type" = TransHeader."Transfer Type"::Loan) THEN
              IF (TransHeader."Salesperson Code" = '') THEN
                ERROR(Text002) ;

            TransferPostReceipt.RUN(TransHeader);
          END;
          //End T20.3
        END;
      END;
    END;

    PROCEDURE AddLotNo@1000000000();
    VAR
      TrnfLineRec@1000000000 : Record 5741;
      LocationRec@1000000001 : Record 14;
      ItemRec@1000000002 : Record 27;
      TrnfLineReserve@1000000003 : Codeunit 99000836;
      ReservEntry@1000000004 : Record 337;
      ItemLedgEntry@1000000005 : Record 32;
      i@1000000006 : Integer;
      ReservEntry2@1000000007 : Record 337;
      tempQty@1000000008 : ARRAY [3] OF Decimal;
    BEGIN
      //T20.2
      TrnfLineRec.RESET;
      TrnfLineRec.SETRANGE(TrnfLineRec."Document No.",TransHeader."No.");
      IF TrnfLineRec.FIND('-') THEN
      REPEAT
        IF LocationRec.GET(TrnfLineRec."Transfer-from Code") AND (LocationRec."Directed Put-away and Pick" = FALSE) AND (TrnfLineRec."Qty. to Ship" > 0) THEN
          IF ItemRec.GET(TrnfLineRec."Item No.") AND (ItemRec."Item Tracking Code" <> '')THEN
          BEGIN
            CASE TransHeader."Transfer Type" OF
              TransHeader."Transfer Type"::Loan, TransHeader."Transfer Type"::Transfer :
                TrnfLineReserve.CallItemTrackingHSL(TrnfLineRec,0);

              TransHeader."Transfer Type"::"Return Loan" :
              BEGIN
                ReservEntry.RESET;
                ReservEntry.LOCKTABLE;

                CLEAR(tempQty);
                tempQty[1] := TrnfLineRec."Quantity (Base)";

                i := 0;
                ReservEntry2.RESET;
                IF ReservEntry2.FINDLAST THEN
                  i := ReservEntry2."Entry No.";

                ItemLedgEntry.RESET;
                ItemLedgEntry.SETRANGE(ItemLedgEntry."Entry Type", ItemLedgEntry."Entry Type"::Transfer);
                ItemLedgEntry.SETRANGE(ItemLedgEntry.Positive,TRUE);
                ItemLedgEntry.SETRANGE(ItemLedgEntry."Item No.", TrnfLineRec."Item No.");
                ItemLedgEntry.SETRANGE(ItemLedgEntry."Location Code", TrnfLineRec."Transfer-from Code");
                ItemLedgEntry.SETRANGE(ItemLedgEntry."Document No.",TrnfLineRec."Source No.");
                ItemLedgEntry.SETRANGE(ItemLedgEntry."Document Line No.",TrnfLineRec."Source Line No.");
                IF ItemLedgEntry.FIND('-') THEN
                REPEAT
                  tempQty[2] := ItemLedgEntry."Remaining Quantity";

                  IF (tempQty[2] > 0) THEN BEGIN
                    tempQty[3] := 0;
                    IF (tempQty[1] <= tempQty[2]) THEN BEGIN
                      tempQty[3] := tempQty[1];
                      tempQty[1] -= tempQty[1];
                    END
                    ELSE BEGIN
                      tempQty[3] := tempQty[2];
                      tempQty[1] -= tempQty[2];
                    END;

                    i += 1;
                    ReservEntry.INIT;
                    ReservEntry."Entry No." := i;
                    ReservEntry."Lot No." := ItemLedgEntry."Lot No.";
                    ReservEntry.Positive := FALSE;
                    ReservEntry."Item No." := TrnfLineRec."Item No.";
                    ReservEntry."Location Code" := TrnfLineRec."Transfer-from Code";
                    ReservEntry.VALIDATE("Quantity (Base)", -tempQty[3]);
                    ReservEntry."Reservation Status" := ReservEntry."Reservation Status"::Surplus;
                    ReservEntry.Description := '';
                    ReservEntry."Creation Date" := TODAY;
                    ReservEntry."Transferred from Entry No." := 0;
                    ReservEntry."Source Type" := 5741;
                    ReservEntry."Source Subtype" := ReservEntry."Source Subtype"::"0";
                    ReservEntry."Source ID" := TrnfLineRec."Document No.";
                    ReservEntry."Source Batch Name" := '';
                    ReservEntry."Source Prod. Order Line" := 0;
                    ReservEntry."Source Ref. No." := TrnfLineRec."Line No.";
                    ReservEntry."Appl.-to Item Entry" := ItemLedgEntry."Entry No.";
                    ReservEntry."Expected Receipt Date" := 0D;
                    ReservEntry."Shipment Date" := TrnfLineRec."Shipment Date";
                    ReservEntry."Created By" := USERID;
                    ReservEntry."Changed By" := '';
                    ReservEntry."Qty. per Unit of Measure" := ItemLedgEntry."Qty. per Unit of Measure";
                    ReservEntry."Suppressed Action Msg." := FALSE;
                    ReservEntry."Planning Flexibility" := ReservEntry."Planning Flexibility"::Unlimited;
                    ReservEntry."Expiration Date" := ItemLedgEntry."Expiration Date";
                    ReservEntry.Correction := FALSE;
                    ReservEntry."Item Tracking" := ReservEntry."Item Tracking"::"Lot No.";
                    ReservEntry.INSERT;


                    i += 1;
                    ReservEntry.INIT;
                    ReservEntry."Entry No." := i;
                    ReservEntry."Lot No." := ItemLedgEntry."Lot No.";
                    ReservEntry.Positive := TRUE;
                    ReservEntry."Item No." := TrnfLineRec."Item No.";
                    ReservEntry."Location Code" := TrnfLineRec."Transfer-to Code";
                    ReservEntry.VALIDATE("Quantity (Base)", tempQty[3]);
                    ReservEntry."Reservation Status" := ReservEntry."Reservation Status"::Surplus;
                    ReservEntry.Description := '';
                    ReservEntry."Creation Date" := TODAY;
                    ReservEntry."Transferred from Entry No." := 0;
                    ReservEntry."Source Type" := 5741;
                    ReservEntry."Source Subtype" := ReservEntry."Source Subtype"::"1";
                    ReservEntry."Source ID" := TrnfLineRec."Document No.";
                    ReservEntry."Source Batch Name" := '';
                    ReservEntry."Source Prod. Order Line" := 0;
                    ReservEntry."Source Ref. No." := TrnfLineRec."Line No.";
                    ReservEntry."Appl.-to Item Entry" := 0;
                    ReservEntry."Expected Receipt Date" := TrnfLineRec."Receipt Date";
                    ReservEntry."Shipment Date" := 0D;
                    ReservEntry."Created By" := USERID;
                    ReservEntry."Changed By" := '';
                    ReservEntry."Qty. per Unit of Measure" := ItemLedgEntry."Qty. per Unit of Measure";
                    ReservEntry."Suppressed Action Msg." := FALSE;
                    ReservEntry."Planning Flexibility" := ReservEntry."Planning Flexibility"::Unlimited;
                    ReservEntry."Expiration Date" := ItemLedgEntry."Expiration Date";
                    ReservEntry.Correction := FALSE;
                    ReservEntry."Item Tracking" := ReservEntry."Item Tracking"::"Lot No.";
                    ReservEntry.INSERT;
                  END;
                UNTIL (ItemLedgEntry.NEXT = 0) OR (tempQty[1] = 0);
              END;
            END;
          END;
      UNTIL TrnfLineRec.NEXT=0;
      //End T20.2
    END;

    BEGIN
    {
      T20.1   12 Okt 2015   SY    Validating Location Code & Ownership
      T20.2   12 Okt 2015   SY    Add codes : add lot no.
      T20.3   12 Okt 2015   SY    Add codes
      T00.01  23 Mar 16     SH    Add codes to fix validation Return Loan before Post Shipment
    }
    END.
  }
