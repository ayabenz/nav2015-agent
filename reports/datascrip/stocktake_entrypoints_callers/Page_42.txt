    LOCAL PROCEDURE ActionPostInvoiceClick@6();
    VAR
      PurchaseHeader@1004 : Record 38;
      ApprovalMgt@1003 : Codeunit 439;
      tCust@1002 : Record 18;
      tmpTotalNilai@1001 : Decimal;
      tSL@1000 : Record 37;
      TextErr01@1005 : TextConst 'ENU=Customer dengan VAT Bus. Posting Group ''BUMN'' dan nilai Amount+VAT %1 10,000,000 \Harus menggunakan Code Transaction ''%2''';
      SlsLine@1007 : Record 37;
      SlsPost@1006 : Codeunit 80;
    BEGIN
      // D34.09
      //CheckDeptMAS();
      //CheckDimLine() -> Table 357 dan table 25000 tidak ada

      CheckVatKurs;

      // BNY -> REQID = SLS 14.16
      IF tCust.GET("Sell-to Customer No.") THEN
        IF (tCust."VAT Bus. Posting Group" = 'BUMN') THEN BEGIN
          tmpTotalNilai := 0;
          tSL.SETRANGE("Document Type", "Document Type");
          tSL.SETRANGE("Document No.", "No.");
          IF tSL.FINDFIRST THEN REPEAT
            IF tSL.Quantity<>0 THEN
            BEGIN
              IF "Currency Factor" <> 0 THEN
                tmpTotalNilai += tSL."Amount Including VAT" / "Currency Factor" * (tSL."Qty. to Invoice"/tSL.Quantity)
              ELSE
                tmpTotalNilai += tSL."Amount Including VAT" *(tSL."Qty. to Invoice"/tSL.Quantity) ;
            END;
          UNTIL tSL.NEXT = 0;
          tmpTotalNilai := ROUND(tmpTotalNilai,0.01);
          IF (tmpTotalNilai <= 10000000) THEN
            BEGIN
              IF "Code Transaction" <> '04' THEN
                ERROR(TextErr01,'<=','04');
            END
          ELSE
            BEGIN
              IF "Code Transaction" <> '03' THEN
                   ERROR(TextErr01,'>','03');
            END;
        END;
      //---------------

      //BLOCK JIKA type Item tapi item no.nya kosong
      IF ("No." <> '') THEN BEGIN
        tSL.RESET;
        tSL.SETRANGE("Document No.", "No.");
        tSL.SETRANGE("Document Type", "Document Type");
        tSL.SETRANGE(Type,tSL.Type::Item);
        IF tSL.FINDFIRST THEN REPEAT
          IF tSL."No." = '' THEN
                ERROR('Item No. masih kosong pada line no '+FORMAT(tSL."Line No."));
        UNTIL tSL.NEXT = 0;
      END;

      //BLOCK JIKA type resource tapi item no.nya kosong
      IF ("No." <> '') THEN BEGIN
        tSL.RESET;
        tSL.SETRANGE("Document No.", "No.");
        tSL.SETRANGE("Document Type", "Document Type");
        tSL.SETRANGE(Type,tSL.Type::Resource);
        IF tSL.FINDFIRST THEN REPEAT
          IF tSL."No." = '' THEN
                ERROR('Resource No. masih kosong pada line no '+FORMAT(tSL."Line No."));
        UNTIL tSL.NEXT = 0;
      END;

      //MPT 16022017 - Block jika "qty to invoice" = 0 namun memiliki nilai "intensif tambahan amount".
      IF ("No." <> '') THEN BEGIN
        tSL.RESET;
        tSL.SETRANGE("Document No.", "No.");
        tSL.SETRANGE("Document Type", "Document Type");
        IF tSL.FINDFIRST THEN REPEAT
          IF (tSL."Qty. to Invoice" = 0) AND (tSL."Insentif Tambahan Line Amount" > 0) THEN
                ERROR('Tidak Bisa Proses Invoice Karena Insentif Tambahan Pada Baris '+FORMAT(tSL."Line No.")+', Quantity to Invoice-nya 0.');
        UNTIL tSL.NEXT = 0;
      END;
      //MPT 16022017 - Block jika "qty to invoice" = 0 namun memiliki nilai "intensif tambahan amount".

      // BNY 20.11.26
      IF ("No." <> '') THEN BEGIN
        tSL.RESET;
        tSL.SETRANGE("Document No.", "No.");
        tSL.SETRANGE("Document Type", "Document Type");
        IF tSL.FINDFIRST THEN REPEAT
          IF (tSL."Qty. to Invoice" > tSL."Quantity Shipped") THEN
                ERROR('"Qty. to Invoice" tidak boleh lebih dari "Qty. Shipped". (Line No. : '+FORMAT(tSL."Line No.")+')');
        UNTIL tSL.NEXT = 0;
      END;
      // BNY 20.11.26

      TESTFIELD("Promised Delivery Date");
      IF ApprovalMgt.PrePostApprovalCheck(Rec, PurchaseHeader) THEN BEGIN
        IF ApprovalMgt.TestSalesPrepayment(Rec) THEN
          ERROR(STRSUBSTNO(Text001,"Document Type","No."))
        ELSE BEGIN
          IF ApprovalMgt.TestSalesPayment(Rec) THEN
            ERROR(STRSUBSTNO(Text002,"Document Type","No."))
          ELSE
            CODEUNIT.RUN(CODEUNIT::"Sales-Post (Y/N) Invoice",Rec);
        END;
      END;
      // D34.09

      //T26.4
      SlsLine.RESET;
      SlsLine.SETRANGE(SlsLine."Document Type", "Document Type");
      SlsLine.SETFILTER(SlsLine."Document No.", "No.");
      SlsLine.SETFILTER(SlsLine."Quantity Invoiced", '<>%1', 0);

      BEGIN
        SlsPost.UpdateSlsChrgRemLineAmt(Rec);
        //T25
        SlsPost.CalcInsentifTambahanLineAmt2(Rec,2); //MPT 23/01/2017 - tipe 2 digunakan pada saat Posting Invoice tiket id : 8845.
        //End T25
      END;
      //End T26.4
    END;

    PROCEDURE LastReceiptDate@12() LastDate : Date;
    VAR
      SalesShipmentHeader@1000 : Record 110;
    BEGIN
      //---start HSL
      SalesShipmentHeader.SETRANGE(SalesShipmentHeader."Order No.","No.");
      //SalesShipmentHeader.SETFILTER(SalesShipmentHeader."Customer Receipt Date",'<>0D');
      {IF SalesShipmentHeader.FINDLAST THEN BEGIN
        LastDate:=SalesShipmentHeader."Customer Receipt Date";
        EXIT(LastDate)
      END;
      }

      IF SalesShipmentHeader.FINDFIRST THEN BEGIN
        LastDate:=SalesShipmentHeader."Customer Receipt Date";
        REPEAT
          IF (SalesShipmentHeader."Customer Receipt Date"<>0D) AND (SalesShipmentHeader."Customer Receipt Date">LastDate) THEN
            LastDate:=SalesShipmentHeader."Customer Receipt Date";
        UNTIL SalesShipmentHeader.NEXT=0;
        EXIT(LastDate)

      END;
      //-------end HSL
    END;
