    PROCEDURE GetRemainingAmount@1000000027(EntryNo@1000000000 : Integer) : Decimal;
    VAR
      LDetCustLedgEntry@1000000001 : Record 379;
      TotalRemainingAmount@1000000002 : Decimal;
    BEGIN
      TotalRemainingAmount := 0;
      LDetCustLedgEntry.RESET;
      LDetCustLedgEntry.SETCURRENTKEY("Cust. Ledger Entry No.","Posting Date");
      LDetCustLedgEntry.SETRANGE("Cust. Ledger Entry No.",EntryNo);
      REPEAT
         TotalRemainingAmount += LDetCustLedgEntry.Amount;
      UNTIL LDetCustLedgEntry.NEXT = 0;
      EXIT(TotalRemainingAmount);
    END;

    PROCEDURE SetLimitRemAmt@1000000001(PLimitRemAmt@1000000000 : Decimal;PLimitRemAmt2@1000000005 : Decimal);
    VAR
      LRemAmtPercentage@1000000001 : Decimal;
      LDetCustLedgEntry@1000000002 : Record 379;
      LAmount@1000000003 : Decimal;
      LRemAmount@1000000004 : Decimal;
    BEGIN
      IF TypeFilter > 0 THEN
      BEGIN
         //MARKEDONLY(FALSE);
         RESET;
         IF FINDSET THEN
         REPEAT
            LAmount := 0;
            LRemAmount := 0;
            LDetCustLedgEntry.RESET;
            LDetCustLedgEntry.SETCURRENTKEY("Cust. Ledger Entry No.","Entry Type","Posting Date");
            LDetCustLedgEntry.SETRANGE("Cust. Ledger Entry No.","Entry No.");
            IF LDetCustLedgEntry.FINDSET THEN
            BEGIN
               REPEAT
                  IF LDetCustLedgEntry."Entry Type" IN [
                  LDetCustLedgEntry."Entry Type"::"Initial Entry",
                  LDetCustLedgEntry."Entry Type"::"Unrealized Loss",
                  LDetCustLedgEntry."Entry Type"::"Unrealized Gain",
                  LDetCustLedgEntry."Entry Type"::"Realized Loss",
                  LDetCustLedgEntry."Entry Type"::"Realized Gain",
                  LDetCustLedgEntry."Entry Type"::"Payment Discount",
                  LDetCustLedgEntry."Entry Type"::"Payment Discount (VAT Excl.)",
                  LDetCustLedgEntry."Entry Type"::"Payment Discount (VAT Adjustment)",
                  LDetCustLedgEntry."Entry Type"::"Payment Tolerance",
                  LDetCustLedgEntry."Entry Type"::"Payment Discount Tolerance",
                  LDetCustLedgEntry."Entry Type"::"Payment Tolerance (VAT Excl.)",
                  LDetCustLedgEntry."Entry Type"::"Payment Tolerance (VAT Adjustment)",
                  LDetCustLedgEntry."Entry Type"::"Payment Discount Tolerance (VAT Excl.)",
                  LDetCustLedgEntry."Entry Type"::"Payment Discount Tolerance (VAT Adjustment)"] THEN
                     LAmount += LDetCustLedgEntry.Amount
                  ELSE IF (LDetCustLedgEntry."Entry Type"=LDetCustLedgEntry."Entry Type"::Application) AND
                  (LDetCustLedgEntry."Document Type"=LDetCustLedgEntry."Document Type"::"Credit Memo")THEN
                     LAmount += LDetCustLedgEntry.Amount;   //tambahan hendra 30 Maret 2010
                  LRemAmount += LDetCustLedgEntry.Amount;
               UNTIL LDetCustLedgEntry.NEXT = 0;

            ELSE

               MARK(TRUE)
            ELSE
            IF (TypeFilter = 3) AND (PLimitRemAmt <= LRemAmtPercentage) AND (LRemAmtPercentage <= PLimitRemAmt2) THEN
               MARK(TRUE)
            ELSE
               MARK(FALSE);
         UNTIL NEXT = 0;
         MARKEDONLY(TRUE);
         CurrPage.UPDATE(FALSE);
      END;
    END;
