OBJECT Codeunit 50084 Sales-Post + Print Invoice
{
  OBJECT-PROPERTIES
  {
    Date=08/06/15;
    Time=01:00:00;
    Modified=Yes;
    Version List=DTS:D34;
  }
  PROPERTIES
  {
    TableNo=36;
    OnRun=BEGIN
            // D34.03
            SalesHeader.COPY(Rec);
            Code;
            Rec := SalesHeader;
            // D34.03
          END;

  }
  CODE
  {
    VAR
      SalesHeader@1007 : Record 36;
      SalesShptHeader@1006 : Record 110;
      SalesInvHeader@1005 : Record 112;
      SalesCrMemoHeader@1004 : Record 114;
      ReturnRcptHeader@1003 : Record 6660;
      ReportSelection@1002 : Record 77;
      SalesPost@1001 : Codeunit 80;
      Selection@1000 : Integer;
      Text000@1010 : TextConst 'ENU=&Invoice';
      Text001@1009 : TextConst 'ENU=Do you want to post and print the %1?';
      Text002@1008 : TextConst 'ENU=&Invoice';

    LOCAL PROCEDURE Code@2();
    BEGIN
      // D34.03
      WITH SalesHeader DO BEGIN
      // YE:050902
        TESTFIELD(Status,Status::Released);
      // YE

        CASE "Document Type" OF
          "Document Type"::Order:
            BEGIN
      { Modified by YE on 15 July 2002
              Selection := STRMENU(Text000,3);
              IF Selection = 0 THEN
                EXIT;
              Ship := Selection IN [1,3];
              Invoice := Selection IN [2,3];
      }
              Selection := STRMENU(Text000,1);
              IF Selection = 0 THEN
                EXIT;
              Ship := FALSE;
              Invoice := Selection IN [1,1];
            END;
          "Document Type"::"Return Order":
            BEGIN
      { Modified by YE on 15 July 2002
              Selection := STRMENU(Text002,3);
              IF Selection = 0 THEN
                EXIT;
              Receive := Selection IN [1,3];
              Invoice := Selection IN [2,3];
      }
              Selection := STRMENU(Text002,1);
              IF Selection = 0 THEN
                EXIT;
              Receive := FALSE;
              Invoice := Selection IN [1,1];
            END ELSE
              IF NOT
                 CONFIRM(
                   Text001,FALSE,
                   "Document Type")
              THEN
                EXIT;
        END;

        SalesPost.RUN(SalesHeader);

        CASE "Document Type" OF
          "Document Type"::Order:
            BEGIN
              IF Ship THEN BEGIN
                SalesShptHeader."No." := "Last Shipping No.";
                SalesShptHeader.SETRECFILTER;
                PrintReport(ReportSelection.Usage::"SM.Order");
              END;
              IF Invoice THEN BEGIN
                SalesInvHeader."No." := "Last Posting No.";
                SalesInvHeader.SETRECFILTER;
                PrintReport(ReportSelection.Usage::"S.Invoice");
              END;
            END;
          "Document Type"::Invoice:
            BEGIN
              IF "Last Posting No." = '' THEN
                SalesInvHeader."No." := "No."
              ELSE
                SalesInvHeader."No." := "Last Posting No.";
              SalesInvHeader.SETRECFILTER;
              PrintReport(ReportSelection.Usage::"S.Invoice");
            END;
          "Document Type"::"Return Order":
            BEGIN
              IF Receive THEN BEGIN
                ReturnRcptHeader."No." := "Last Return Receipt No.";
                ReturnRcptHeader.SETRECFILTER;
                PrintReport(ReportSelection.Usage::"SM.Invoice");
              END;
              IF Invoice THEN BEGIN
                SalesCrMemoHeader."No." := "Last Posting No.";
                SalesCrMemoHeader.SETRECFILTER;
                PrintReport(ReportSelection.Usage::"S.Cr.Memo");
              END;
            END;
          "Document Type"::"Credit Memo":
            BEGIN
              IF "Last Posting No." = '' THEN
                SalesCrMemoHeader."No." := "No."
              ELSE
                SalesCrMemoHeader."No." := "Last Posting No.";
              SalesCrMemoHeader.SETRECFILTER;
              PrintReport(ReportSelection.Usage::"S.Cr.Memo");
            END;
        END;
      END;
      // D34.03
    END;

    LOCAL PROCEDURE PrintReport@1(ReportUsage@1000 : Integer);
    BEGIN
      // D34.03
      ReportSelection.RESET;
      ReportSelection.SETRANGE(Usage,ReportUsage);
      ReportSelection.FIND('-');
      REPEAT
        ReportSelection.TESTFIELD("Report ID");
        CASE ReportUsage OF
          ReportSelection.Usage::"S.Invoice":
            REPORT.RUN(ReportSelection."Report ID",FALSE,FALSE,SalesInvHeader);
          ReportSelection.Usage::"S.Cr.Memo":
            REPORT.RUN(ReportSelection."Report ID",FALSE,FALSE,SalesCrMemoHeader);
          ReportSelection.Usage::"SM.Order":
            REPORT.RUN(ReportSelection."Report ID",FALSE,FALSE,SalesShptHeader);
          ReportSelection.Usage::"SM.Invoice":
            REPORT.RUN(ReportSelection."Report ID",FALSE,FALSE,ReturnRcptHeader);
        END;
      UNTIL ReportSelection.NEXT = 0;
      // D34.03
    END;

    BEGIN
    {

    }
    END.
  }
