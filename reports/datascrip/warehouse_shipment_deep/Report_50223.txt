                               IF tmpWSH.COUNT = 0 THEN CurrReport.BREAK;

                               CLEAR(cWPS);

                               FOR N := 1 TO 10 DO tmpFilterNo[N] := '';

                               tmpFlag := FALSE;
                               tmpInt := 1;

                               tmpWSH.RESET;
                               IF tmpWSH.FIND('-') THEN
                               REPEAT
                                 IF STRLEN(tmpFilterNo[tmpInt]) > 1000 THEN tmpInt += 1;

                                 IF tmpFlag THEN
                                   tmpFilterNo[tmpInt] += '|'
                                 ELSE
                                   tmpFlag := TRUE;

                                 tmpFilterNo[tmpInt] += FORMAT(tmpWSH."No.");

                                 tWSL.RESET;
                                 tWSL.SETRANGE("No.", tmpWSH."No.");
                                 IF tWSL.FIND('-') THEN
                                 BEGIN
                                   cWPS.SetPostingSettings(FALSE); //-- INVOICE = FALSE
                                   cWPS.SetPrint(FALSE);
                                   cWPS.SetSkipOpenDoc(TRUE);
                                   cWPS.SetCustomProcess(RF_PostingDate,TRUE);
                                   cWPS.RUN(tWSL);
                                   CLEAR(cWPS);
                                 END;
                               UNTIL tmpWSH.NEXT = 0;

                               IF Print AND ((tmpFilterNo[1]+tmpFilterNo[2]+tmpFilterNo[3]) <> '') THEN
                               BEGIN
                                 //Sales Shipment Header
                                 tSSH.RESET;
                                 tSSH.SETFILTER("Shipment No.",
                                    (tmpFilterNo[1]+tmpFilterNo[2]+
                                     tmpFilterNo[3]+tmpFilterNo[4]+
                                     tmpFilterNo[5]+tmpFilterNo[6]+
                                     tmpFilterNo[7]+tmpFilterNo[8]+
                                     tmpFilterNo[9]+tmpFilterNo[10]));
                                 IF tSSH.FIND('-') THEN

                                  //IF (tSSH."Sell-to Customer No." = '39310') AND (tSSH."Shortcut Dimension 1 Code"='364') THEN

                                   //REPORT.RUN(REPORT::"Sales - Shipment",FALSE,FALSE,tSSH);
                                   REPORT.RUN(REPORT::"Report Delivery Order",FALSE,FALSE,tSSH);
                                   //End T34.1

                                 //Return Shipment Header
                                 tRSH.RESET;
                                 tRSH.SETFILTER("Shipment No.",
                                    (tmpFilterNo[1]+tmpFilterNo[2]+
                                     tmpFilterNo[3]+tmpFilterNo[4]+
                                     tmpFilterNo[5]+tmpFilterNo[6]+
                                     tmpFilterNo[7]+tmpFilterNo[8]+
                                     tmpFilterNo[9]+tmpFilterNo[10]));
                                 IF tRSH.FIND('-') THEN
                                 REPORT.RUN(REPORT::"Purchase - Return Shipment",FALSE,FALSE,tRSH);

                                 //Transfer Shipment Header
                                 tTSH.RESET;
                                 tTSH.SETFILTER("Shipment No.",
                                    (tmpFilterNo[1]+tmpFilterNo[2]+
                                     tmpFilterNo[3]+tmpFilterNo[4]+
                                     tmpFilterNo[5]+tmpFilterNo[6]+
                                     tmpFilterNo[7]+tmpFilterNo[8]+
                                     tmpFilterNo[9]+tmpFilterNo[10]));
                                 IF tTSH.FIND('-') THEN
                                 REPORT.RUN(REPORT::"Transfer Shipment",FALSE,FALSE,tTSH);

                               END;

                               tmpWSH.DELETEALL;
                               //cWPS.GetResultMessage;
                               MESSAGE(Text1001);
                               CLEAR(cWPS);
                             END;
