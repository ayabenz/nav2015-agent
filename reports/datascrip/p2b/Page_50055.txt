               END;

    OnInsertRecord=BEGIN
                     gRec_P2BHdr.GET("Document No.");
                     gRec_P2BHdr.TESTFIELD(gRec_P2BHdr.Status, gRec_P2BHdr.Status::Open);
                   END;

    OnModifyRecord=BEGIN
                     gRec_P2BHdr.GET("Document No.");
                     gRec_P2BHdr.TESTFIELD(gRec_P2BHdr.Status, gRec_P2BHdr.Status::Open);
                   END;

    OnDeleteRecord=BEGIN
                     gRec_P2BHdr.GET("Document No.");
                     gRec_P2BHdr.TESTFIELD(gRec_P2BHdr.Status, gRec_P2BHdr.Status::Open);
                   END;

    OnAfterGetCurrRecord=VAR
                           Bol_Allow@1000 : Boolean;
                         BEGIN
                           gRec_P2BHdr.GET("Document No.");
                           gRec_UserSetup.GET(USERID);

                           IF NOT gRec_RespCenter.GET(gRec_UserSetup."P2B Resp. Ctr. Filter") THEN
                             Bol_Allow := TRUE
                           ELSE IF gRec_RespCenter."P2B Job Title" = gRec_RespCenter."P2B Job Title"::"Sales Dept." THEN
                             Bol_Allow := TRUE
                           ELSE
                             Bol_Allow := FALSE;

                           {IF P2BManagement.IsOwner(USERID,gRec_P2BHdr.Owner) OR Bol_Allow THEN BEGIN
                             gBol_QtyEditable := TRUE;
                           END ELSE BEGIN
                             IF IsBackOffice OR IsKPC THEN
                               gBol_QtyEditable := FALSE
                             ELSE
                               gBol_QtyEditable := TRUE;

                             IF (P2BManagement.IsCrossDivisi(gRec_P2BHdr))
                             AND NOT(P2BManagement.IsOwner(USERID,gRec_P2BHdr.Owner))
                             AND NOT(P2BManagement.IsSalesAdmin(USERID)) THEN
                             BEGIN
                               gBol_QtyEditable := FALSE;
                             END;
                           END;   }

                           gBol_QtyEditable := TRUE;//FDR 1 Mar 16

                           IF gRec_P2BHdr."Posting To" = gRec_P2BHdr."Posting To"::"Item Journal" THEN
                             gBol_DisableLocation := TRUE
                           ELSE
                             gBol_DisableLocation := FALSE;
                         END;
