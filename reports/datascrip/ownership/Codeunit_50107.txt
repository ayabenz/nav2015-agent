    PROCEDURE BlockStocktakePurchaseHeader@5(PurchaseHeader@1003 : Record 38);
    VAR
      tblStocktake@1002 : Record 50063;
      PurchaseLine@1001 : Record 39;
      tblItem@1004 : Record 27;
    BEGIN
      IF (PurchaseHeader."No." <> '') THEN BEGIN

        PurchaseLine.RESET;
        PurchaseLine.SETRANGE("Document No.", PurchaseHeader."No.");
        PurchaseLine.SETRANGE("Document Type", PurchaseHeader."Document Type");
        PurchaseLine.SETRANGE(Type, PurchaseLine.Type::Item);

        IF NOT PurchaseLine.FINDFIRST THEN EXIT;

        tblStocktake.RESET;
        tblStocktake.SETFILTER("Start Date", '<=%1', CURRENTDATETIME);
        tblStocktake.SETFILTER("End Date", '>=%1', CURRENTDATETIME);

        IF PurchaseHeader."Document Type" IN [PurchaseHeader."Document Type"::Order, PurchaseHeader."Document Type"::Invoice] THEN
            tblStocktake.SETFILTER("Jenis Transaksi", 'IN|ALL')
        ELSE IF PurchaseHeader."Document Type" IN [PurchaseHeader."Document Type"::"Credit Memo", PurchaseHeader."Document Type"::"Return Order"] THEN
            tblStocktake.SETFILTER("Jenis Transaksi", 'OUT|ALL');

        IF tblStocktake.FINDFIRST THEN
          REPEAT
            PurchaseLine.SETRANGE("Item Category Code");
            PurchaseLine.SETRANGE("Product Group Code");
            PurchaseLine.SETRANGE("No.");

            PurchaseLine.SETFILTER("Location Code", tblStocktake."Location Code");
            IF PurchaseLine.FINDFIRST AND ((tblStocktake."Item Category Code" <> '') OR (tblStocktake."Product Group Code" <> '') OR (tblStocktake."Item No." <> '')) THEN
              BEGIN
                IF (tblStocktake."Block Type" = tblStocktake."Block Type"::"Item Category") AND (tblStocktake."Item Category Code" <> '') THEN
                  PurchaseLine.SETFILTER("Item Category Code", tblStocktake."Item Category Code")
                ELSE IF (tblStocktake."Block Type" = tblStocktake."Block Type"::"Product Group") AND (tblStocktake."Product Group Code" <> '') THEN
                  PurchaseLine.SETFILTER("Product Group Code", tblStocktake."Product Group Code")
                ELSE IF (tblStocktake."Block Type" = tblStocktake."Block Type"::Item) AND (tblStocktake."Item No." <> '') THEN
                  PurchaseLine.SETFILTER("No.", tblStocktake."Item No.");
                IF PurchaseLine.FINDFIRST THEN
                  ERROR(Text001, FORMAT(PurchaseLine."No."));
              END;
          UNTIL tblStocktake.NEXT = 0;

      END;
    END;

    PROCEDURE BlockStocktakeTransferHeader@2("Transfer Header"@1003 : Record 5740;JenisTransaksi@1005 : Code[3]);
    VAR
      tblStocktake@1002 : Record 50063;
      TransferHeader@1000 : Record 5740;
      TransferLine@1001 : Record 5741;
      tblItem@1004 : Record 27;
    BEGIN
      TransferHeader.RESET;
      TransferHeader.SETFILTER("No.", "Transfer Header"."No.");
      IF NOT TransferHeader.FINDFIRST THEN EXIT;

      IF (TransferHeader."No." <> '') THEN BEGIN

        TransferLine.RESET;
        TransferLine.SETRANGE("Document No.", TransferHeader."No.");
        IF NOT TransferLine.FINDFIRST THEN EXIT;

        tblStocktake.RESET;
        tblStocktake.SETFILTER("Start Date", '<=%1', CURRENTDATETIME);
        tblStocktake.SETFILTER("End Date", '>=%1', CURRENTDATETIME);
        tblStocktake.SETFILTER("Jenis Transaksi", 'ALL|'+JenisTransaksi);

        IF tblStocktake.FINDFIRST THEN
          REPEAT
            TransferLine.SETRANGE("Item Category Code");
            TransferLine.SETRANGE("Product Group Code");
            TransferLine.SETRANGE("Item No.");

            IF tblStocktake."Jenis Transaksi" IN [tblStocktake."Jenis Transaksi"::OUT,tblStocktake."Jenis Transaksi"::ALL] THEN
              BEGIN
                TransferLine.SETRANGE("Transfer-to Code");
                TransferLine.SETFILTER("Transfer-from Code", tblStocktake."Location Code");
              END;

            IF NOT TransferLine.FINDFIRST THEN
             IF tblStocktake."Jenis Transaksi" IN [tblStocktake."Jenis Transaksi"::"IN",tblStocktake."Jenis Transaksi"::ALL] THEN
              BEGIN
                TransferLine.SETRANGE("Transfer-from Code");
                TransferLine.SETFILTER("Transfer-to Code", tblStocktake."Location Code");
              END;

            IF TransferLine.FINDFIRST AND ((tblStocktake."Item Category Code" <> '') OR (tblStocktake."Product Group Code" <> '') OR (tblStocktake."Item No." <> '')) THEN
              BEGIN
                IF (tblStocktake."Block Type" = tblStocktake."Block Type"::"Item Category") AND (tblStocktake."Item Category Code" <> '') THEN
                  TransferLine.SETFILTER("Item Category Code", tblStocktake."Item Category Code")
                ELSE IF (tblStocktake."Block Type" = tblStocktake."Block Type"::"Product Group") AND (tblStocktake."Product Group Code" <> '') THEN
                  TransferLine.SETFILTER("Product Group Code", tblStocktake."Product Group Code")
                ELSE IF (tblStocktake."Block Type" = tblStocktake."Block Type"::Item) AND (tblStocktake."Item No." <> '') THEN
                  TransferLine.SETFILTER("Item No.", tblStocktake."Item No.");
                IF TransferLine.FINDFIRST THEN
                  ERROR(Text001, FORMAT(TransferLine."Item No."));
              END;
          UNTIL tblStocktake.NEXT = 0;

      END;
    END;

    PROCEDURE BlockStocktakeWarehouseReceiptHeader@3(WhsReceiptHeader@1003 : Record 7316);
    VAR
      tblStocktake@1002 : Record 50063;
      WhsReceiptLine@1001 : Record 7317;
      tblItem@1004 : Record 27;
    BEGIN
      IF (WhsReceiptHeader."No." <> '') THEN BEGIN

        WhsReceiptLine.RESET;
        WhsReceiptLine.SETRANGE("No.", WhsReceiptHeader."No.");
        IF NOT WhsReceiptLine.FINDFIRST THEN EXIT;

        tblStocktake.RESET;
        tblStocktake.SETFILTER("Start Date", '<=%1', CURRENTDATETIME);
        tblStocktake.SETFILTER("End Date", '>=%1', CURRENTDATETIME);
        tblStocktake.SETFILTER("Jenis Transaksi", 'IN|ALL');

        IF tblStocktake.FINDFIRST THEN
          REPEAT
            WhsReceiptLine.SETFILTER(WhsReceiptLine."Location Code", tblStocktake."Location Code");
            IF WhsReceiptLine.FINDFIRST AND ((tblStocktake."Item Category Code" <> '') OR (tblStocktake."Product Group Code" <> '') OR (tblStocktake."Item No." <> '')) THEN
              BEGIN
                tblItem.GET(WhsReceiptLine."Item No.");
                IF (tblStocktake."Block Type" = tblStocktake."Block Type"::"Item Category") AND (tblStocktake."Item Category Code" <> '') THEN
                  tblItem.SETFILTER("Item Category Code", tblStocktake."Item Category Code")
                ELSE IF (tblStocktake."Block Type" = tblStocktake."Block Type"::"Product Group") AND (tblStocktake."Product Group Code" <> '') THEN
                  tblItem.SETFILTER("Product Group Code", tblStocktake."Product Group Code")
                ELSE IF (tblStocktake."Block Type" = tblStocktake."Block Type"::Item) AND (tblStocktake."Item No." <> '') THEN
                  tblItem.SETFILTER("No.", tblStocktake."Item No.");
                IF tblItem.FINDFIRST THEN
                  REPEAT
                    IF tblItem."No." = WhsReceiptLine."Item No." THEN
                      ERROR(Text001, FORMAT(tblItem."No."));
                  UNTIL tblItem.NEXT = 0;
              END;
          UNTIL tblStocktake.NEXT = 0;

      END;
    END;

    PROCEDURE ReplaceStr@4(Text@1000 : Text[250];OldStr@1001 : Text[250];NewStr@1002 : Text[250]) Result : Text[250];
    BEGIN
      WHILE STRPOS(Text, OldStr) > 0 DO
        Text := DELSTR(Text,STRPOS(Text, OldStr)) + NewStr + COPYSTR(Text,STRPOS(Text, OldStr) + STRLEN(OldStr));
      Result := Text;
    END;

    BEGIN
    END.
