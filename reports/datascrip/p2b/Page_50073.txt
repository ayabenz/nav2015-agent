    PROCEDURE GetNoWhseShipment@1000000000(VAR NoShipment@1000000004 : Code[20];VAR StatusDok@1000000003 : Text[30]) : Code[20];
    VAR
      Tsh@1000000000 : Record 5744;
      Th@1000000001 : Record 5740;
      WSL@1000000002 : Record 7321;
      WSH@1000000005 : Record 7320;
    BEGIN
      Tsh.SETRANGE(Tsh."P2B Document No.","No.");
      IF Tsh.FINDFIRST THEN BEGIN
        NoShipment:=Tsh."Shipment No.";
        StatusDok:=FORMAT(Tsh."Truck Status")
      END
      ELSE IF Th.GET("Posted Doc No.") THEN BEGIN
        WSL.SETRANGE(WSL."Source Document",WSL."Source Document"::"Outbound Transfer");
        WSL.SETRANGE(WSL."Source No.",Th."No.");
        IF WSL.FINDLAST THEN BEGIN
          WSH.GET(WSL."No.");
          WSH.CALCFIELDS(WSH."Completely Picked");
          NoShipment:=WSL."No.";
          IF WSH."Completely Picked" THEN
            StatusDok:='Completely Picked'
          ELSE IF WSH."Pick Created" THEN
            StatusDok:='Pick Created'
          ELSE
            StatusDok:='Belum diPick';
        END;
      END;
    END;

    BEGIN
    END.
