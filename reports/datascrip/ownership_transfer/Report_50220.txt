    PROCEDURE CalcNeed@1(pItem@1000 : Record 27;ItemCategoryFilter@1000000000 : Text[250];LocationFilter@1001 : Text[250];OwnerFilter@1000000001 : Text[250];VariantFilter@1002 : Text[250];ReorderPoint@1003 : Decimal);
    VAR
      n@1000000002 : Integer;
    BEGIN
      // D08.01
      WITH pItem DO
      BEGIN
        { DVZ - Filter KPC }
        IF cuUSM.GetUserKpcStatus THEN
        BEGIN
          FILTERGROUP := 2;
          SETFILTER("Global Dimension 1 Filter", cuUSM.GetUserKpcDeptCode);
          SETFILTER("Global Dimension 2 Filter", cuUSM.GetUserKpcAreaCode);
          SETFILTER("Location Filter", cuUSM.SetLocationFilterUserKpc);
          FILTERGROUP := 0;
        END;


        IF (strLoc <> '') THEN BEGIN
          FILTERGROUP := 2;
          SETFILTER("Location Filter", strLocAll);
          SETFILTER("Owner Filter", strOwnAll);
          FILTERGROUP := 0;
        END;

        { DVZ }
        FOR n := 1 TO 5 DO KolomDec[n] := 0;


        { DVZ - WMS }
        IF (ItemCategoryFilter <> '') THEN SETFILTER("Item Category Code", ItemCategoryFilter);
        IF (OwnerFilter <> '') THEN SETFILTER("Owner Filter", OwnerFilter);
        SETFILTER("Location Filter",LocationFilter);
        SETFILTER("Variant Filter",VariantFilter);

        CALCFIELDS(

          "Qty. on Sales Order"
          ,"Trans. Ord. Shipment (Qty.)"
          ,"Qty. On Purchase Return"
          ,"Reserved Qty. on Inventory"
          ,"Qty. On Purchase Return"
          ,"Qty. On Item Journal"
          ,"Qty. on Service Order"

          { DVZ }
          ,"Qty P2B Line (Other)"
          ,"Qty P2B Line (Item Journal)"
          ,"Qty. On Ownshp. Transfer"
          //,"Qty. On Ownshp. Ledg. Entry"
          );

        SETRANGE("Date Filter",0D,12319999D);

        IF USERID='ad' THEN
          MESSAGE('%1 %2',strLocAll ,strOwnAll);

        KolomDec[2] := "Qty. on Service Order"
                      + "Qty. on Sales Order"
      //                + "Qty. on Kit Sales Lines"
                      + "Scheduled Need (Qty.)"
                      + "Planning Issues (Qty.)"
                      + "Trans. Ord. Shipment (Qty.)"
                      + "Qty. On Purchase Return"
                      + GetQtyItemJournal("No.",LocationFilter,OwnerFilter,FALSE)
                      + ("Qty P2B Line (Other)" + "Qty P2B Line (Item Journal)")
                      //+ ("Qty. On Ownshp. Transfer" * (-1))
                      + ("Qty. On Ownshp. Transfer")
                      ;


        KolomDec[3] := AvailToPromise.CalcScheduledReceipt(pItem)
                      + GetQtyItemJournal("No.",LocationFilter,OwnerFilter,TRUE)
                      ;

        CALCFIELDS(Inventory,"Qty. On Ownshp. Ledg. Entry");

        KolomDec[1] := "Qty. On Ownshp. Ledg. Entry";

        KolomDec[4] := "Qty. On Ownshp. Ledg. Entry"
                      + KolomDec[3] //+ ScheduledReceipt
                      - KolomDec[2] //- GrossRequirement
                      ;
        IF USERID='ad' THEN BEGIN
           MESSAGE('%1 %2 %3',FORMAT("Qty. On Ownshp. Ledg. Entry"),FORMAT(KolomDec[3]),FORMAT(KolomDec[2]));
        END;

        KolomDec[5] := KolomDec[1] - KolomDec[2] ;

        InvtReorder := ProjAvailBalance < ReorderPoint;

        IF (OwnerFilter <> '') THEN
        BEGIN
          SETFILTER("Owner Filter", '=%1', '');
          CALCFIELDS(
            "Qty. on Service Order"
            ,"Qty. on Sales Order"
            ,"Scheduled Need (Qty.)"
            ,"Planning Issues (Qty.)"
            ,"Trans. Ord. Shipment (Qty.)"
            ,"Qty. On Purchase Return"
            ,"Qty P2B Line (Other)"
            ,"Qty P2B Line (Item Journal)");

          CurrentBlankOwnerGross :=
            "Qty. on Service Order"
            + "Qty. on Sales Order"
            + "Scheduled Need (Qty.)"
            + "Planning Issues (Qty.)"
            + "Trans. Ord. Shipment (Qty.)"
            + "Qty. On Purchase Return"
            + GetQtyItemJournal("No.",LocationFilter,'#$#$#$',FALSE)
            + ("Qty P2B Line (Other)" + "Qty P2B Line (Item Journal)")
            ;
            IF (USERID='AD')
              THEN
            BEGIN
              MESSAGE('=SERVO%1=SO=%2=SN=%3==PI==%4=TOS=%5=QPR=%6=IJ=%7=P2B=%8 ',
                "Qty. on Service Order"
                , "Qty. on Sales Order"
                , "Scheduled Need (Qty.)"
                , "Planning Issues (Qty.)"
                , "Trans. Ord. Shipment (Qty.)"
                , "Qty. On Purchase Return"
                , GetQtyItemJournal("No.",LocationFilter,'#$#$#$',FALSE)
                , ("Qty P2B Line (Other)" + "Qty P2B Line (Item Journal)")
               );
            END;
         END;
      END;
      // D08.01
    END;

    PROCEDURE GetQtyItemJournal@1000000002(PItemNo@1000000003 : Code[20];PLocationCode@1000000002 : Text[250];POwnerFilter@1000000001 : Text[250];PMinusValue@1000000000 : Boolean) RESULT : Decimal;
    VAR
      tmpIJL@1000000004 : Record 83;
    BEGIN
      // D08.01
      { DVZ }
      tmpIJL.RESET;
      tmpIJL.SETCURRENTKEY("Journal Template Name","Entry Type","Quantity (Base)","Item No.","Location Code");
      tmpIJL.SETFILTER("Journal Template Name",'<>%1', 'PHYS. INVE');
      IF PMinusValue THEN
        tmpIJL.SETFILTER("Quantity (Base)",'<0')
      ELSE
        tmpIJL.SETFILTER("Quantity (Base)",'>0');
      tmpIJL.SETFILTER("Entry Type", '=%1', tmpIJL."Entry Type"::"Negative Adjmt.");
      tmpIJL.SETFILTER("Item No.", PItemNo);
      IF PLocationCode <> '' THEN
        tmpIJL.SETFILTER("Location Code", PLocationCode)
      ELSE
        tmpIJL.SETRANGE("Location Code");

      IF POwnerFilter  = '#$#$#$strloc' THEN
      BEGIN
        tmpIJL.SETFILTER(Owner, '=%1', '');
        IF tmpIJL.FINDFIRST THEN
        REPEAT
          RESULT += tmpIJL."Quantity (Base)";
        UNTIL tmpIJL.NEXT = 0;
      END
      ELSE
      BEGIN
        IF POwnerFilter <> '' THEN
          tmpIJL.SETFILTER(Owner, POwnerFilter)
        ELSE
          tmpIJL.SETRANGE(Owner);

        IF tmpIJL.CALCSUMS("Quantity (Base)") THEN
          RESULT := tmpIJL."Quantity (Base)";
      END;

      IF PMinusValue THEN RESULT := RESULT * (-1);
      // D08.01
    END;

    BEGIN

    }
    END.
