    PROCEDURE GetOptionPostingTo@1000000003(PDocNo@1000000001 : Code[20]) : Integer;
    VAR
      LP2BHeader@1000000000 : Record 50013;
      LResponsibilityCenter@1000000002 : Record 5714;
    BEGIN
      LP2BHeader.RESET;
      LP2BHeader.SETFILTER("No.",PDocNo);
      IF LP2BHeader.FINDFIRST THEN
      BEGIN
         gBol_QtyOnHandVisible := FALSE;
         //CurrForm.QtyOnHandTextBox.VISIBLE :=   FALSE;
      //      (LP2BHeader.Requester<>LP2BHeader.Owner) AND NOT(P2BManagement.IsCrossDivisi(LP2BHeader));
         IF (LResponsibilityCenter.GET(P2BManagement.GetUserP2BRespCenter(USERID))) THEN
            PemilikBarang := LResponsibilityCenter."P2B Resp. Owner"
         ELSE
            PemilikBarang := FALSE;
         IF LP2BHeader."Posting To" = LP2BHeader."Posting To"::"Item Journal" THEN
            LocationHeader := "Location Code"+' '+Owner
         ELSE
            LocationHeader := LP2BHeader."From Location"+' '+LP2BHeader."From Location Ownership";
         IF LResponsibilityCenter.GET(LP2BHeader.Requester) THEN
           ToLocation := CONVERTSTR(LResponsibilityCenter."P2B Location Code",',','|');
         EXIT(LP2BHeader."Posting To");
      END
      ELSE
      BEGIN
         LocationHeader := '';
         ToLocation := '';
         EXIT(-1);
      END;
    END;

    PROCEDURE HitungAvailable@1000000006();
    VAR
      Item@1000000000 : Record 27;
    BEGIN
      //IF ("Item No." <> '')AND(GetOptionPostingTo("Document No.") = 0) THEN
      GetOptionPostingTo("Document No.");

      IF ("Item No." <> '') THEN
      BEGIN
         TextLocationHeader := LocationHeader;
         IF TextLocationHeader= '' THEN TextLocationHeader := "Location Code";
         QtyAvailable := P2BManagement.GetQuantityAvailable("Item No.",TextLocationHeader);
         Item.GET("Item No.");
         WITH Item DO BEGIN
            RESET;
            SETFILTER("Location Filter",ToLocation);
            CALCFIELDS(Inventory);
            QtyOnHand := Inventory;
         END;
      END
      ELSE
      BEGIN
         QtyAvailable := 0;
         QtyOnHand := 0;
      END;
    END;
