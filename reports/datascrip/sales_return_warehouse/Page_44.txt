    LOCAL PROCEDURE CheckDimensionSalesLine@14();
    VAR
      SalesCrMemoLine@1001 : Record 37;
      DimSetEntry@1000 : Record 480;
      Text001@1005 : TextConst 'ENU=Department Code (%1) tidak sesuai dengan Dimension (%2)';
      Text003@1004 : TextConst 'ENU=Salesperson Code (%1) tidak sesuai dengan Dimension (%2)';
      Text002@1003 : TextConst 'ENU=Product Group Code (%1) tidak sesuai dengan Dimension (%2)';
    BEGIN
      //
      IF USERID IN ['DATASCRIP\BENNY_AZIZ','DATASCRIP\TEGUH_FAJAR']  THEN
      BEGIN

      WITH SalesCrMemoLine DO
        BEGIN
          RESET;
          SETFILTER("Document Type", 'Credit Memo');
          SETFILTER("Document No.", "No.");
          IF FINDFIRST THEN
            REPEAT
              BEGIN
                DimSetEntry.RESET;
                DimSetEntry.SETRANGE("Dimension Set ID", "Dimension Set ID");
                IF DimSetEntry.FINDFIRST THEN
                  REPEAT
                    BEGIN
                      IF DimSetEntry."Dimension Code" = 'DEPARTMENT' THEN
                        IF DimSetEntry."Dimension Value Code" <> Rec."Shortcut Dimension 1 Code" THEN
                          ERROR(Text001, Rec."Shortcut Dimension 1 Code", DimSetEntry."Dimension Value Code");

                      IF DimSetEntry."Dimension Code" = 'PRODUCTGROUP' THEN
                        IF DimSetEntry."Dimension Value Code" <> "Product Group Code" THEN
                          ERROR(Text002, "Product Group Code", DimSetEntry."Dimension Value Code");

                      IF DimSetEntry."Dimension Code" = 'SLSPERSON' THEN
                        IF DimSetEntry."Dimension Value Code" <> Rec."Salesperson Code" THEN
                          ERROR(Text003, Rec."Salesperson Code", DimSetEntry."Dimension Value Code");
                    END;
                  UNTIL DimSetEntry.NEXT = 0;
              END;
            UNTIL NEXT = 0;
        END;
      END;
      //
    END;
