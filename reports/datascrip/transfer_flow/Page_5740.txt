    PROCEDURE CheckOwnership@1000000000();
    VAR
      tLoc@1000000000 : Record 14;
      tTransLine@1000000001 : Record 5741;
    BEGIN
      //T13.01
      TESTFIELD("Transfer-from Code");
      tLoc.GET("Transfer-from Code");
      IF tLoc.Ownership THEN
        IF "From Location Ownership" = '' THEN BEGIN
          tTransLine.RESET;
          tTransLine.SETRANGE("Document No.", "No.");
          tTransLine.SETFILTER(Quantity, '<>0');
          IF tTransLine.FINDFIRST THEN REPEAT
            tTransLine.TESTFIELD(Owner);
          UNTIL tTransLine.NEXT = 0;
        END ELSE
          TESTFIELD("From Location Ownership");

      TESTFIELD("Transfer-to Code");
      tLoc.GET("Transfer-to Code");
      IF tLoc.Ownership THEN
        IF "To Location Ownership" = '' THEN BEGIN
          tTransLine.RESET;
          tTransLine.SETRANGE("Document No.", "No.");
          tTransLine.SETFILTER(Quantity, '<>0');
          IF tTransLine.FINDFIRST THEN REPEAT
            tTransLine.TESTFIELD(Owner);
          UNTIL tTransLine.NEXT = 0;
        END ELSE
          TESTFIELD("To Location Ownership");
      //T13.01
    END;

    LOCAL PROCEDURE GetLastWHShptNo@2();
    BEGIN

      //T20.01
      IF ("No." <> '') THEN BEGIN
        gRec_WHShptHdr.RESET;
        gRec_WHShptHdr.SETRANGE("Line Source Document", gRec_WHShptHdr."Line Source Document"::"Outbound Transfer");
        gRec_WHShptHdr.SETRANGE("Line Source No.", "No.");
        IF gRec_WHShptHdr.FINDLAST THEN
          gCod_WhseShipmentNo := gRec_WHShptHdr."No."
        ELSE
          gCod_WhseShipmentNo := '';

        IF (gCod_WhseShipmentNo = '') THEN BEGIN
          gRec_TransShptHdr.RESET;
          gRec_TransShptHdr.SETCURRENTKEY("Transfer Order No.");
          gRec_TransShptHdr.SETRANGE("Transfer Order No.", "No.");
          IF gRec_TransShptHdr.FINDLAST THEN
            gCod_WhseShipmentNo := gRec_TransShptHdr."Shipment No.";
        END;

      END ELSE
        gCod_WhseShipmentNo := '';
      //T20.01
    END;
