    PROCEDURE CekPostOwnTransfer@1000000005(VAR Rec_OwnershipTransfer@1000000000 : Record 50037) : Boolean;
    VAR
      Rec_OLE@1003 : Record 50035;
      Rec_Location@1000000001 : Record 14;
      Rec_OwnershipTrans@1000 : Record 50037;
      Dec_QtyFrom@1001 : Decimal;
      Dec_QtyTo@1002 : Decimal;
    BEGIN
      //T30.01
      WITH Rec_OwnershipTransfer DO BEGIN
        FINDSET;
        REPEAT
          //TESTFIELD("Journal Batch Name");
          //TESTFIELD("Line No.");
          TESTFIELD("Item No.");
          TESTFIELD("Posting Date");
          TESTFIELD("Document No.");
          //TESTFIELD(Description);
          TESTFIELD("Location Code");

          Rec_Location.GET("Location Code");
          IF Rec_Location.Ownership THEN
            Rec_Location.TESTFIELD(Ownership,TRUE);

          TESTFIELD(Quantity);
          //T30.02
          TESTFIELD(Owner);
          //TESTFIELD("Transfer-from Owner");
          //TESTFIELD("Transfer-to Owner");
          IF ("Transfer-from Owner" <> '') AND ("Transfer-to Owner" <> '') THEN
            ERROR('Transfer-from Owner dan Transfer-to Owner hanya boleh diisi salah satu.');
          //T30.02

          IF "P2B Document No." <> '' THEN BEGIN
            Rec_OLE.RESET;
            Rec_OLE.SETCURRENTKEY("P2B Doc No.");
            Rec_OLE.SETRANGE("P2B Doc No.", "P2B Document No.");
            IF Rec_OLE.FINDFIRST THEN
              ERROR('Ownership Ledger Entry dengan No. P2B %1 sudah pernah diposting.', "P2B Document No.");
          END;

          //Check Balance Before Posting
          Dec_QtyFrom := 0;
          Dec_QtyTo := 0;
          IF "Transfer-from Owner" <> '' THEN
            Dec_QtyFrom := Quantity
          ELSE IF "Transfer-to Owner" <> '' THEN
            Dec_QtyTo := Quantity;

          Rec_OwnershipTrans.RESET;
          Rec_OwnershipTrans.SETRANGE("Journal Batch Name", "Journal Batch Name");
          Rec_OwnershipTrans.SETFILTER("Line No.", '<>%1', "Line No.");
          Rec_OwnershipTrans.SETRANGE("Item No.", "Item No.");
          Rec_OwnershipTrans.SETRANGE("Document No.", "Document No.");
          Rec_OwnershipTrans.SETRANGE("Posting Date", "Posting Date");
          Rec_OwnershipTrans.SETFILTER("Transfer-from Owner", '<>%1', '');
          Rec_OwnershipTrans.SETRANGE("Transfer-to Owner", '');
          IF Rec_OwnershipTrans.FINDSET THEN
            REPEAT
              Dec_QtyFrom := Dec_QtyFrom + Rec_OwnershipTrans.Quantity;
            UNTIL Rec_OwnershipTrans.NEXT = 0;

          Rec_OwnershipTrans.SETRANGE("Transfer-from Owner", '');
          Rec_OwnershipTrans.SETFILTER("Transfer-to Owner", '<>%1', '');
          IF Rec_OwnershipTrans.FINDSET THEN
            REPEAT
              Dec_QtyTo := Dec_QtyTo + Rec_OwnershipTrans.Quantity;
            UNTIL Rec_OwnershipTrans.NEXT = 0;

          IF Dec_QtyFrom <> Dec_QtyTo THEN
            ERROR('Quantity must be Balance for Item %1 Doc No. %2 Posting Date %3',Rec_OwnershipTrans."Item No.",
              Rec_OwnershipTrans."Document No.",Rec_OwnershipTrans."Posting Date");

        UNTIL NEXT = 0;
      END;
      EXIT(TRUE);
      //T30.01
    END;

      tSL@1000000002 : Record 37;

      tmpDate@1000000004 : Date;

      // D34.02
      //Default date untuk lot no. dan expiration date
      tmpDate := 102612D;

      tWRL.RESET;
      tWRL.SETRANGE("No.", tWRH."No.");
      IF tWRL.FIND('-') THEN

        IF (tWRL."Source Type" = 37) THEN //SALES LINE - RETURN RECEIPT

          tSL.SETRANGE("Document No.", tWRL."Source No.");
          tSL.SETRANGE("Line No.", tWRL."Source Line No.");

          BEGIN
            IF (tSL."Appl.-from Item Entry" <> 0) THEN
            BEGIN
              tItem.GET(tWRL."Item No.");

                tILE.RESET;
                tILE.SETRANGE("Entry No.", tSL."Appl.-from Item Entry");

                BEGIN
                  IF (tILE."Lot No." = '') AND (tILE."Expiration Date" = 0D) THEN
                  BEGIN
                    tILE."Lot No." := FORMAT(tmpDate, 0, '<year4>/<month,2>/<day,2>');
                    tILE."Expiration Date" := CALCDATE('+3Y',tmpDate);
                    tILE.MODIFY;
                  END;

                  tRE.RESET;
                  tRE.SETRANGE("Source Type", tWRL."Source Type");
                  tRE.SETRANGE("Source Subtype", tWRL."Source Subtype");
                  tRE.SETRANGE("Source ID", tWRL."Source No.");
                  tRE.SETRANGE("Source Ref. No.", tWRL."Source Line No.");
                  IF tRE.FIND('-') THEN
                  BEGIN
                    tRE."Lot No." := tILE."Lot No.";
                    tRE."Expiration Date" := tILE."Expiration Date";
                    tRE."Appl.-from Item Entry" := tSL."Appl.-from Item Entry";
                    tRE.MODIFY;
                  END;
                END;
              END;
            END;
          END;
        END;
      UNTIL tWRL.NEXT = 0;
      // D34.02
    END;

      Q_OwnershipTransfer@1000000980 : Decimal;
    BEGIN
      // D34.02
      { DVZ - Cek Quantity Available }
      { ********************************************** }

      {
      Option pTransType :
      -----------------------
      0. Blank
      1. Sales Line
      2. Purchase Line
      3. Transfer Line
      4. Item Journal Line ( tidak aktif )
      5. P2B Line
      6. Ownership Transfer
      -----------------------
      }
      { ********************************************** }

      Result := FALSE;

      //T30.03 tLocation.GET(pLocationCode);
      IF tLocation.GET(pLocationCode) THEN; //T30.03
      IsLocOwnership := tLocation.Ownership;

      IF IsLocOwnership
        AND (pOwner = '' )
        AND NOT SkipBlankOwner
      THEN
        ERROR('Owner cannot be blank');

      IF NOT IsLocOwnership THEN
      BEGIN
        { I. Item ledger Entry - Quantity }

        tItem.RESET;
        tItem.GET(pItemNo);
        tItem.SETRANGE("Location Filter", pLocationCode);
        tItem.CALCFIELDS(
          Inventory
          ,"Qty. on Sales Order"
          ,"Trans. Ord. Shipment (Qty.)"
          ,"Qty. On Purchase Return"
          ,"Qty P2B Line (Item Journal)"
          ,"Qty P2B Line (Other)"
          ,"Qty. On Ownshp. Transfer"
          );

        Q_ItemLedgerEntry := tItem.Inventory;
      END
      ELSE
      BEGIN
        { II. Ownership ledger Entry - Quantity }

        tItem.RESET;
        tItem.GET(pItemNo);
        tItem.SETRANGE("Location Filter", pLocationCode);
        IF NOT SkipBlankOwner THEN
          tItem.SETRANGE("Owner Filter", pOwner);

        tItem.CALCFIELDS(
          "Qty. On Ownshp. Ledg. Entry"
          ,"Qty. on Sales Order"
          ,"Trans. Ord. Shipment (Qty.)"
          ,"Qty. On Purchase Return"
          ,"Qty P2B Line (Item Journal)"
          ,"Qty P2B Line (Other)"
          ,"Qty. On Ownshp. Transfer"
          );

        Q_OwnershipLedgerEntry := tItem."Qty. On Ownshp. Ledg. Entry";
      END;

      { III. Sales Line - Quantity (Order) }
      Q_SalesLine := tItem."Qty. on Sales Order";
      IF (pTransType = 1) THEN
          Q_SalesLine := Q_SalesLine - pxQuantity;


      { IV. Purchase Line - Quantity (Return Order) }
      Q_PurchaseLine := tItem."Qty. On Purchase Return";
      IF (pTransType = 2) THEN
          Q_PurchaseLine := Q_PurchaseLine - pxQuantity;


      { ** Service Invoice Line - Quantity (dikosongkan) }
      { ** Loaner - Quantity (dikosongkan) }

      { V. Transfer Line - Quantity }
      Q_TransferLine := tItem."Trans. Ord. Shipment (Qty.)";
      IF (pTransType = 3) THEN
          Q_TransferLine := Q_TransferLine - pxQuantity;


      { VI. Item Journal Line - Quantity }
      tIJL.RESET;
      tIJL.SETCURRENTKEY("Journal Template Name","Entry Type","Item No.","Location Code",Owner,"Quantity (Base)");
      tIJL.SETFILTER("Journal Template Name",'<>%1&<>%2', 'PHYS. INVE', 'REVALUATIO');
      tIJL.SETRANGE("Entry Type", tIJL."Entry Type"::"Negative Adjmt.");
      tIJL.SETRANGE("Item No.", pItemNo);
      tIJL.SETRANGE("Location Code", pLocationCode);
      IF IsLocOwnership THEN
        IF NOT SkipBlankOwner THEN
          tIJL.SETRANGE(Owner, pOwner);

      { Diff. with Item Availability, Here }
      //tIJL.SETFILTER("Journal Template Name",'<>%1','PHYS. INVE');
      //tIJL.SETFILTER("Journal Template Name",'<>%1&<>%2', 'PHYS. INVE', 'REVALUATIO');

      tIJL.SETFILTER("Quantity (Base)",'>0');
      { ---------- }
      //IF tIJL.FIND('-') THEN
      IF tIJL.FINDFIRST THEN
      BEGIN
        IF tIJL.CALCSUMS("Quantity (Base)") THEN
        Q_ItemJournalLine := tIJL."Quantity (Base)";
        //Q_ItemJournalLine := -tIJL."Quantity (Base)" - pxQuantity;
      END;
      IF (pTransType = 4) THEN
        //IF Q_ItemJournalLine = 0 THEN
          //Q_ItemJournalLine := pxQuantity
        //ELSE
        Q_ItemJournalLine := Q_ItemJournalLine - pxQuantity;


      { VIIa. P2B - Quantity }
      Q_P2BLine := tItem."Qty P2B Line (Item Journal)" + tItem."Qty P2B Line (Other)";
      IF (pTransType = 5) THEN
        //IF Q_P2BLine = 0 THEN
          //Q_P2BLine := pxQuantity
        //ELSE
        Q_P2BLine := Q_P2BLine - pxQuantity;


      { VIII. Rec - UoM - Quantity Base }
      tIUoM.RESET;
      tIUoM.SETRANGE("Item No.", pItemNo);
      tIUoM.SETRANGE(Code, pUoMCode);
      IF tIUoM.FINDFIRST THEN
        Q_Base := (pQuantity * tIUoM."Qty. per Unit of Measure")
      ELSE
        ERROR('Unit of Measure Code invalid ...');

      { IX. xRec - UoM - Quantity Base }
      tIUoM.RESET;
      tIUoM.SETRANGE("Item No.", pItemNo);
      tIUoM.SETRANGE(Code, pxUoMCode);
      IF tIUoM.FINDFIRST THEN
        Q_xBase := (pxQuantity * tIUoM."Qty. per Unit of Measure")
      ELSE
        ERROR('Unit of Measure Code invalid ...');


      { X. Ownership Transfer }
      //Q_OwnershipTransfer := (tItem."Qty. On Ownshp. Transfer" * (-1));
      Q_OwnershipTransfer := tItem."Qty. On Ownshp. Transfer";
      IF (pTransType = 6) THEN
        //IF Q_OwnershipTransfer = 0 THEN
          //Q_OwnershipTransfer := pxQuantity
        //ELSE
        Q_OwnershipTransfer := Q_OwnershipTransfer - pxQuantity;
        //Q_OwnershipTransfer := Q_OwnershipTransfer - pQuantity;


      { Formula }
      { ********************************************** }

      IF ((Q_Base - Q_xBase) <= 0) THEN
        {
        Result := ((Q_ItemLedgerEntry
                + Q_OwnershipLedgerEntry
                - Q_SalesLine
                - Q_PurchaseLine
                - Q_TransferLine
                - Q_ItemJournalLine
                - Q_P2BLine
                - Q_OwnershipTransfer
                - Q_Base + Q_xBase) >= 0)
        }
        Result := TRUE { Qty lebih kecil or sama dengan qty lama }
      ELSE
        IF ((Q_Base - Q_xBase) > 0)

            + Q_OwnershipLedgerEntry

            - Q_TransferLine
            - Q_ItemJournalLine
            - Q_P2BLine

        ELSE

              + Q_OwnershipLedgerEntry
              - Q_SalesLine
              - Q_PurchaseLine
              - Q_TransferLine
              - Q_ItemJournalLine
              - Q_P2BLine
              - Q_OwnershipTransfer
              - Q_Base) < 0)
          THEN
            Result := FALSE; { Qty lebih besar dari qty lama dan stok tidak cukup }

      vQtyAvailable := Q_ItemLedgerEntry
        + Q_OwnershipLedgerEntry
        - Q_SalesLine
        - Q_PurchaseLine
        - ABS(Q_TransferLine)
        - Q_ItemJournalLine
        - Q_P2BLine
        - Q_OwnershipTransfer;

      //T37.01
      // jika "Allow Negative Availability" di thick, maka qty in diperhitungkan
      IF (tItem."Allow Negative Availability") AND (pTransType = 1) THEN BEGIN

        //T37.02           tItem."Qty. on Assembly Order"+tItem."Qty. On Ownshp. Transfer";

        IF vQtyAvailable >= 0 THEN
          Result := TRUE
        ELSE
          Result := FALSE;
      END;
      //T37.01

      // D34.02
    END;

    LOCAL PROCEDURE CreateBMPHeader@1140005(VAR poutBmpHeader@1000000000 : OutStream;pintCols@1000000001 : Integer;pintRows@1000000002 : Integer;pintSize@1000000004 : Integer;pblnVertical@1000000006 : Boolean);
    VAR
      charInf@1000000003 : Char;
      lintResolution@1000000005 : Integer;
      lintWidth@1000000007 : Integer;
      lintHeight@1000000008 : Integer;
    BEGIN

      lintResolution := ROUND(2835 / pintSize,1,'=');

      IF pblnVertical THEN BEGIN
        lintWidth  := pintRows * pintSize;
        lintHeight := pintCols;
      END ELSE BEGIN
        lintWidth  := pintCols * pintSize;
        lintHeight := pintRows * pintSize;
      END;

      charInf := 'B' ;
      poutBmpHeader.WRITE(charInf, 1);
      charInf := 'M' ;
      poutBmpHeader.WRITE(charInf, 1);
      poutBmpHeader.WRITE(54 + pintRows * pintCols * 3, 4); //SIZE BMP
      poutBmpHeader.WRITE(0, 4); //APPLICATION SPECIFIC
      poutBmpHeader.WRITE(54, 4); //OFFSET DATA PIXELS
      poutBmpHeader.WRITE(40, 4); //NUMBER OF BYTES IN HEADER FROM THIS POINT
      poutBmpHeader.WRITE(lintWidth, 4); //WIDTH PIXEL
      poutBmpHeader.WRITE(lintHeight, 4); //HEIGHT PIXEL
      poutBmpHeader.WRITE(65536 * 24 + 1, 4); //COLOR DEPTH
      poutBmpHeader.WRITE(0, 4); //NO. OF COLOR PANES & BITS PER PIXEL
      poutBmpHeader.WRITE(0, 4); //SIZE BMP DATA
      poutBmpHeader.WRITE(lintResolution, 4); //HORIZONTAL RESOLUTION
      poutBmpHeader.WRITE(lintResolution, 4); //VERTICAL RESOLUTION
      poutBmpHeader.WRITE(0, 4); //NO. OF COLORS IN PALETTE
      poutBmpHeader.WRITE(0, 4); //IMPORTANT COLORS
    END;

    LOCAL PROCEDURE CreateBarcodeDetail@1000000014(pintLines@1000000005 : Integer;pintBars@1000000006 : Integer;pintSize@1000000009 : Integer;pblnVertical@1000000010 : Boolean;VAR poutBmpHeader@1000000007 : OutStream);
    VAR
      lintLineLoop@1000000004 : Integer;
      lintBarLoop@1000000003 : Integer;
      ltxtByte@1000000002 : Text[1];
      lchar@1000000001 : Char;
      lintChainFiller@1000000000 : Integer;
      lintSize@1000000008 : Integer;
      lintCount@1000000011 : Integer;
    BEGIN
      IF pblnVertical THEN BEGIN
        FOR lintBarLoop := 1 TO (bxtBarcodeBinary.LENGTH) DO BEGIN

          FOR lintLineLoop := 1 TO (pintLines * pintSize) DO BEGIN
            bxtBarcodeBinary.GETSUBTEXT(ltxtByte,lintBarLoop,1);

            IF ltxtByte = '1' THEN
              lchar := 0
            ELSE
              lchar := 255;

            poutBmpHeader.WRITE(lchar,1);
            poutBmpHeader.WRITE(lchar,1);
            poutBmpHeader.WRITE(lchar,1);
          END;

          FOR lintChainFiller := 1 TO (lintLineLoop MOD 4) DO BEGIN
            //Adding 0 bytes if needed - line end
            lchar := 0;
            poutBmpHeader.WRITE(lchar,1);
          END;
        END;
      END ELSE BEGIN
        FOR lintLineLoop := 1 TO pintLines * pintSize DO BEGIN
          FOR lintBarLoop := 1 TO bxtBarcodeBinary.LENGTH DO BEGIN
            bxtBarcodeBinary.GETSUBTEXT(ltxtByte,lintBarLoop,1);

            IF ltxtByte = '1' THEN
              lchar := 0
            ELSE
              lchar := 255;

            FOR lintSize := 1 TO pintSize DO BEGIN
              //Putting Pixel: Black or White
              poutBmpHeader.WRITE(lchar,1);
              poutBmpHeader.WRITE(lchar,1);
              poutBmpHeader.WRITE(lchar,1);
            END
          END;

          FOR lintChainFiller := 1 TO ((lintBarLoop* pintSize) MOD 4) DO BEGIN
            //Adding 0 bytes if needed - line end
            lchar := 0;
            poutBmpHeader.WRITE(lchar,1);
          END;
        END;
      END
    END;

        AND NOT SkipBlankOwner
      THEN
        ERROR('Owner cannot be blank');

      IF NOT IsLocOwnership THEN

        { I. Item ledger Entry - Quantity }

        tItem.RESET;

          ,"Qty. on Sales Order"
          ,"Trans. Ord. Shipment (Qty.)"
          ,"Qty. On Purchase Return"
          ,"Qty P2B Line (Item Journal)"
          ,"Qty P2B Line (Other)"
          ,"Qty. On Ownshp. Transfer"
          );

        Q_ItemLedgerEntry := tItem.Inventory;
      END
      ELSE
      BEGIN
        { II. Ownership ledger Entry - Quantity }

        tItem.RESET;
        tItem.GET(pItemNo);
        tItem.SETRANGE("Location Filter", pLocationCode);
        IF NOT SkipBlankOwner THEN
          tItem.SETRANGE("Owner Filter", pOwner);

        tItem.CALCFIELDS(
          "Qty. On Ownshp. Ledg. Entry"
          ,"Qty. on Sales Order"
          ,"Trans. Ord. Shipment (Qty.)"
          ,"Qty. On Purchase Return"
          ,"Qty P2B Line (Item Journal)"
          ,"Qty P2B Line (Other)"
          ,"Qty. On Ownshp. Transfer"
          );

        Q_OwnershipLedgerEntry := tItem."Qty. On Ownshp. Ledg. Entry";
      END;

      { III. Sales Line - Quantity (Order) }
      Q_SalesLine := tItem."Qty. on Sales Order";

      { IV. Purchase Line - Quantity (Return Order) }
      Q_PurchaseLine := tItem."Qty. On Purchase Return";

      { ** Loaner - Quantity (dikosongkan) }

      { V. Transfer Line - Quantity }
      Q_TransferLine := tItem."Trans. Ord. Shipment (Qty.)";

      tIJL.SETCURRENTKEY("Journal Template Name","Entry Type","Item No.","Location Code",Owner,"Quantity (Base)");
      tIJL.SETFILTER("Journal Template Name",'<>%1&<>%2', 'PHYS. INVE', 'REVALUATIO');

      { VIIa. P2B - Quantity }
      Q_P2BLine := tItem."Qty P2B Line (Item Journal)" + tItem."Qty P2B Line (Other)";

      { VIII. Rec - UoM - Quantity Base }
      tIUoM.RESET;
      tIUoM.SETRANGE("Item No.", pItemNo);
      tIUoM.SETRANGE(Code, pUoMCode);
      IF tIUoM.FIND('-') THEN
        Q_Base := (pQuantity * tIUoM."Qty. per Unit of Measure")
      ELSE
        ERROR('Unit of Measure Code invalid ...');

      { IX. xRec - UoM - Quantity Base }
      tIUoM.RESET;
      tIUoM.SETRANGE("Item No.", pItemNo);
      tIUoM.SETRANGE(Code, pxUoMCode);
      IF tIUoM.FIND('-') THEN
        Q_xBase := (pxQuantity * tIUoM."Qty. per Unit of Measure")
      ELSE
        ERROR('Unit of Measure Code invalid ...');

      { X. Ownership Transfer }
      //Q_OwnershipTransfer := (tItem."Qty. On Ownshp. Transfer" * (-1));
      Q_OwnershipTransfer := tItem."Qty. On Ownshp. Transfer";

      IF (pxLocationCode=pLocationCode) AND (pxOwner=pOwner) THEN
        xTmpQty := pxQuantity
      ELSE
        xTmpQty := 0;

      IF ((pxLocationCode=pLocationCode) AND (pxOwner=pOwner)) THEN
      BEGIN
        IF ((Q_Base - Q_xBase) <= 0) THEN
          Result := TRUE { Qty lebih kecil or sama dengan qty lama }
        ELSE
          IF ((Q_Base - Q_xBase) > 0)
            AND ((
              //xTmpQty +
                Q_ItemLedgerEntry
              + Q_OwnershipLedgerEntry
              - Q_SalesLine
              - Q_PurchaseLine
              - Q_TransferLine
              + Q_ItemJournalLine
              - Q_P2BLine
              - Q_OwnershipTransfer
              - Q_Base) >= 0)
          THEN
            Result := TRUE { Qty lebih besar dari qty lama dan stok cukup }
          ELSE
            IF ((Q_Base - Q_xBase) > 0)
              AND ((
                //xTmpQty +
                  Q_ItemLedgerEntry
                + Q_OwnershipLedgerEntry
                - Q_SalesLine
                - Q_PurchaseLine
                - Q_TransferLine
                + Q_ItemJournalLine
                - Q_P2BLine
                - Q_OwnershipTransfer
                - Q_Base) < 0)
            THEN
              Result := FALSE; { Qty lebih besar dari qty lama dan stok tidak cukup }
      END
      ELSE
        Result := ((Q_ItemLedgerEntry
          + Q_OwnershipLedgerEntry
          - Q_SalesLine
          - Q_PurchaseLine
          - Q_TransferLine
          + Q_ItemJournalLine
          - Q_P2BLine
          - Q_OwnershipTransfer
          - Q_Base) >=0);

      vQtyAvailable := Q_ItemLedgerEntry
        + Q_OwnershipLedgerEntry
        - Q_SalesLine
        - Q_PurchaseLine
        - Q_TransferLine
        + Q_ItemJournalLine
        - Q_P2BLine
        - Q_OwnershipTransfer;

    END;
