              IF tSL.FINDFIRST THEN REPEAT
                IF tLoc.GET(tSL."Location Code") THEN
                  IF tLoc.Ownership THEN
                    tSL.TESTFIELD(Owner);
              UNTIL tSL.NEXT=0;
            END;

            tWRL."Source Document"::"Purchase Order" : BEGIN
              tPL.RESET;
              tPL.SETRANGE("Document Type", tPL."Document Type"::Order);
              tPL.SETRANGE("Document No.", tWRL."Source No.");
              tPL.SETRANGE(Type, tPL.Type::Item);
              IF tPL.FINDFIRST THEN REPEAT
                IF tLoc.GET(tPL."Location Code") THEN
                  IF tLoc.Ownership THEN
                    tPL.TESTFIELD(Owner);
              UNTIL tPL.NEXT=0;
            END;
          END;
        UNTIL tWRL.NEXT = 0;
      END;

    END;

          tblAPT.SETRANGE(SeqNo, tblWRL."Line No.");

            tblAPT.Code20_1   := tblWRL."No.";

          END;
        END;
      UNTIL tblWRL.NEXT = 0;

    END;

    PROCEDURE GetDocType@1000000001() : Integer;
    VAR
      docType@1000000000 : Integer;
    BEGIN
      docType := -1;
      TmpWhseRcvLine.RESET;
      TmpWhseRcvLine.SETFILTER(TmpWhseRcvLine."No.", "No.");
      IF NOT TmpWhseRcvLine.FIND('-') THEN
        ERROR('Warehouse Receipt Line Is Empty')
      ELSE
      BEGIN
        CASE TmpWhseRcvLine."Source Document" OF
          TmpWhseRcvLine."Source Document"::"Purchase Order": docType := 1;
          TmpWhseRcvLine."Source Document"::"Sales Return Order": docType := 2;
          TmpWhseRcvLine."Source Document"::"Inbound Transfer": docType := 3;
          TmpWhseRcvLine."Source Document"::"Sales Order": docType := 4;
        END;
      END;
      EXIT(docType);
    END;
