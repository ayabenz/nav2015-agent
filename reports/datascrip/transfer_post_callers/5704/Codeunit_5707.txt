OBJECT Codeunit 5707 TransferOrder-Post + Print
{
  OBJECT-PROPERTIES
  {
    Date=02/15/16;
    Time=08:35:51;
    Modified=Yes;
    Version List=NAVW17.00;
  }
  PROPERTIES
  {
    TableNo=5740;
    OnRun=BEGIN
            TransHeader.COPY(Rec);
            Code;
            Rec := TransHeader;
          END;

  }
  CODE
  {
    VAR
      Text000@1000 : TextConst 'ENU=&Ship,&Receive';
      TransHeader@1001 : Record 5740;
      TransShptHeader@1002 : Record 5744;
      TransRcptHeader@1003 : Record 5746;
      tTL@1005 : Record 5741;
      tItem@1004 : Record 27;

    LOCAL PROCEDURE Code@2();
    VAR
      TransLine@1000 : Record 5741;
      DefaultNumber@1001 : Integer;
      TransferPostShipment@1002 : Codeunit 5704;
      TransferPostReceipt@1003 : Codeunit 5705;
      Selection@1004 : ' ,Shipment,Receipt';
    BEGIN
      WITH TransHeader DO BEGIN


       {IF "No." <> '' THEN BEGIN
          tTL.RESET;
          tTL.SETRANGE("Document No.", "No.");
          IF tTL.FINDFIRST THEN REPEAT
             tItem.RESET;
             tItem.SETRANGE("No.",tTL."Item No.");
             tItem.SETRANGE("Global Dimension 1 Code",'102');
             IF tItem.FINDFIRST THEN
                ERROR('Item ' + FORMAT(tTL."Item No.")
                          + ' sedang tidak dapat digunakan, '
                          + 'karena sedang dalam proses stocktake');

            {
              IF tItem.GET(tTL."Item No.") THEN
                IF tItem."Item Category Code" IN
                  ['104-70000'] THEN
                    ERROR('Item ' + FORMAT(tTL."Item No.")
                        + ' sedang tidak dapat digunakan, '
                        + 'karena sedang dalam proses stocktake');
            }
          UNTIL tTL.NEXT = 0;
        END; }


        TransLine.SETRANGE("Document No.","No.");
        IF TransLine.FIND('-') THEN
          REPEAT
            IF (TransLine."Quantity Shipped" < TransLine.Quantity) AND
               (DefaultNumber = 0)
            THEN
              DefaultNumber := 1;
            IF (TransLine."Quantity Received" < TransLine.Quantity) AND
               (DefaultNumber = 0)
            THEN
              DefaultNumber := 2;
          UNTIL (TransLine.NEXT = 0) OR (DefaultNumber > 0);

        IF DefaultNumber = 0 THEN
          DefaultNumber := 1;
        Selection := STRMENU(Text000,DefaultNumber);
        CASE Selection OF
          0:
            EXIT;
          1:
            BEGIN
              TransferPostShipment.RUN(TransHeader);
              TransShptHeader."No." := "Last Shipment No.";
              TransShptHeader.SETRECFILTER;
              TransShptHeader.PrintRecords(FALSE);
            END;
          2:
            BEGIN
              TransferPostReceipt.RUN(TransHeader);
              TransRcptHeader."No." := "Last Receipt No.";
              TransRcptHeader.SETRECFILTER;
              TransRcptHeader.PrintRecords(FALSE);
            END;
        END;
      END;
    END;

    BEGIN
    END.
  }
}
