    PROCEDURE IsRespCenterHasNoLocation@1000000069(PRespCenter@1000000000 : Code[10];PVariant@1000000003 : Code[50];VAR PRangeLocationCode@1000000004 : Code[50]) : Boolean;
    VAR
      LVi_Counter@1000000001 : Integer;
      LRespCode@1000000005 : Code[10];
      LVariantCode@1000000006 : Code[50];
      LVi_Index@1000000007 : Integer;
      LRespCenter@1000000002 : Record 5714;
    BEGIN
      // D34.03
      FOR LVi_Counter := 1 TO ARRAYLEN(RespCenterHasNoLocation) DO
      BEGIN
         IF RespCenterHasNoLocation[LVi_Counter] <> '' THEN
         BEGIN
            LRespCode := SELECTSTR(1,RespCenterHasNoLocation[LVi_Counter]);
            LVariantCode := SELECTSTR(2,RespCenterHasNoLocation[LVi_Counter]);
            IF (PRespCenter = LRespCode) AND (LVariantCode = PVariant) THEN
            BEGIN
               LVi_Index := STRLEN(LRespCode) + STRLEN(LVariantCode) + 3;

      {
             PRangeLocationCode:= COPYSTR(RespCenterHasNoLocation[LVi_Counter],LVi_Index,STRLEN(RespCenterHasNoLocation[LVi_Counter]));
               EXIT(TRUE); }

      //BEGIN FDR 2 May 2011 {1}
               IF PRangeLocationCode = '' THEN
                  PRangeLocationCode := COPYSTR(RespCenterHasNoLocation[LVi_Counter],
                  LVi_Index,STRLEN(RespCenterHasNoLocation[LVi_Counter]))
               ELSE
                  PRangeLocationCode := PRangeLocationCode +',' +COPYSTR(RespCenterHasNoLocation[LVi_Counter],
                  LVi_Index,STRLEN(RespCenterHasNoLocation[LVi_Counter]));
      //END   FDR 2 May 2011 {1}

            END;
         END;
      END;

      //BEGIN FDR 2 May 2011 {2}
      IF PRangeLocationCode<>'' THEN
        EXIT(TRUE)
      ELSE  //END FDR 2 May 2011 {2}
        EXIT(FALSE);
      // D34.03
    END;

    PROCEDURE CheckingHeader@1000000043(VAR Rec@1000000000 : Record 50013);
    VAR
      LLocation@1000000001 : Record 14;
    BEGIN
      WITH Rec DO
      BEGIN
         CASE "Posting To" OF
            "Posting To"::"Item Journal":
               BEGIN
                  TESTFIELD("Journal Template Name");
                  TESTFIELD("Journal Batch Name");
                  TESTFIELD("Gen. Bus. Posting Group");
               END;
            "Posting To"::"Transfer Order":
               BEGIN
                  TESTFIELD("From Location");
                  IF (LLocation.GET("From Location"))AND(LLocation.Ownership) THEN
                     TESTFIELD("From Location Ownership");
                  TESTFIELD("To Location");
                  IF (LLocation.GET("To Location"))AND(LLocation.Ownership) THEN
                     TESTFIELD("To Location Ownership");
                  TESTFIELD("Transit Location");
               END;
            "Posting To"::Loan:
               BEGIN
                  TESTFIELD("From Location");
                  IF (LLocation.GET("From Location"))AND(LLocation.Ownership) THEN
                     TESTFIELD("From Location Ownership");
                  TESTFIELD("To Location");
                  IF (LLocation.GET("To Location"))AND(LLocation.Ownership) THEN
                     TESTFIELD("To Location Ownership");
                  TESTFIELD("Transit Location");
                  TESTFIELD("Loan Type");
                  TESTFIELD("Department Code");
                  TESTFIELD("Salesperson Code");
                  TESTFIELD("Loan Due Date");
                  IF "Loan Type"="Loan Type"::Customer THEN
                    TESTFIELD("Loan-to Customer No.")
               END;
         END;
      END;
    END;

        LP2BLog.RESET;

    PROCEDURE Posted@1000000019(VAR Rec@1000000000 : Record 50013);
    VAR
      WMS@1000000002 : Boolean;

      BEGIN
         TESTFIELD(Status,Status::Release);
         TESTFIELD("Document Type");
         TESTFIELD("Warehouse Code");
         TESTFIELD(Requester);
         TESTFIELD(Owner);
         TESTFIELD("User ID Requester");
         CheckingHeader(Rec);
         CheckingLine("No.",0);

               TESTFIELD("Loan-to Customer No.")
            ELSE IF "Loan Type"="Loan Type"::Department THEN
               TESTFIELD(Rec."Purchaser Code");
         END;

         CASE "Posting To" OF
            "Posting To"::"Item Journal":
               BEGIN
                  //PostingItemJournal(Rec);
                  //FDR Modif WMS
                  CheckLocationType("No.",WMS,LocationCode);
                  IF WMS THEN
                  BEGIN
                    CheckingLine("No.",2);
                    PostingTransferOrderWMS(Rec,LocationCode);
                  END
                  ELSE
                    PostingItemJournal(Rec);
               END;
            "Posting To"::"Transfer Order":
               BEGIN
                  //PostingTransferOrder(Rec);
                  //BEGIN FDR 9 juni 2011
                  IF LocationWMS("From Location") AND LocationWMS("To Location") AND ("From Location"="To Location") THEN
                    PostingOwnerTransfer(Rec)
                  ELSE
                    PostingTransferOrder(Rec);
                  //END FDR 9 juni 2011
               END;
            "Posting To"::Loan:
               BEGIN
                  PostingLoan(Rec);
               END;
         END;
         Window.OPEN(
            '#1#################################\\' +
            'Posting lines              #2######');
         Window.UPDATE(1,'Posting P2B');

         ProcessWorkflow(Rec);
         "Current Step" := TextStepFinish;
         "Current Steps" := TextStepFinish;

         Status := Status::Close;
         UpdateStatusP2BLine("No.",Status);
         Rec."Posting Date" := TODAY;
         MODIFY;
         COMMIT;
         Window.CLOSE;
         MESSAGE('Document P2B No. : %1 posted succesfully. Posted Doc No. : %2', "No.", Rec."Posted Doc No.");
      END;
    END;

    PROCEDURE CheckingLine@1000000033(PDocNo@1000000000 : Code[20];POption@1000000002 : 'Quantity,Location,Owner');

    PROCEDURE HasStepSubstitution@1000000068(PStep@1000000000 : Code[50];VAR PResult@1000000004 : Code[50]) : Boolean;
    VAR
      LVi_Counter@1000000001 : Integer;
      LStepSubstitution@1000000002 : Code[50];
      LResultSubstitution@1000000003 : Code[50];
    BEGIN
      IF STRPOS(PStep,' ') > 0 THEN
        PStep := COPYSTR(PStep,1,STRPOS(PStep,' ')-1);
      FOR LVi_Counter := 1 TO ARRAYLEN(StepSubstitution) DO
      BEGIN
         IF StepSubstitution[LVi_Counter] <> '' THEN
         BEGIN
            LStepSubstitution := StepSubstitution[LVi_Counter];
            LStepSubstitution := COPYSTR(LStepSubstitution,1,STRPOS(LStepSubstitution,'=')-1);
            IF PStep = LStepSubstitution THEN
            BEGIN
               PResult := COPYSTR(StepSubstitution[LVi_Counter],
               STRPOS(StepSubstitution[LVi_Counter],'=')+1,STRLEN(StepSubstitution[LVi_Counter]));
               EXIT(TRUE);
            END;
         END;
      END;
      EXIT(FALSE);
    END;

    PROCEDURE GetHistoryRecordByUserID@3(VAR TmpP2BHeader@1000 : TEMPORARY Record 50013);

               TmpP2BHeader.INIT;
               TmpP2BHeader.TRANSFERFIELDS(LP2BHeader);
               TmpP2BHeader.INSERT;
            END

         UNTIL LP2BHeader.NEXT=0;
      END;

    END;
