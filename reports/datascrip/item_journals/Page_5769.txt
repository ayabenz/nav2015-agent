    PROCEDURE ShowReceiveAppliedForm@1000000000(RcvNo@1000000000 : Code[20]);
    VAR
      tmpPenerimaanLine@1000000001 : Record 50032;
      PgPenerimaanApplied@1000000002 : Page 50095;
      tmpRcvApplyDetail@1000000004 : Record 50033;
      temporaryPenerimaan@1000000003 : TEMPORARY Record 50032;
      hasApplyRcv@1000 : Boolean;
      tmpRcvApplyDetail2@1001 : Record 50033;
    BEGIN
      {**************************************************************************
      * ADD ON BIA
      * 30/07/2010
      **************************************************************************}

      // INIT TEMPORARY RECORD
      temporaryPenerimaan.RESET;
      temporaryPenerimaan.DELETEALL;

      // READ DATA FROM PENERIMAAN LINE
      tmpPenerimaanLine.RESET;
      tmpPenerimaanLine.SETFILTER("No.", RcvNo);
      tmpPenerimaanLine.SETRANGE("Item No.", "Item No.");
      IF tmpPenerimaanLine.FIND('-') THEN
      REPEAT
        temporaryPenerimaan.INIT;
        temporaryPenerimaan := tmpPenerimaanLine;

        temporaryPenerimaan."Qty. On Apply" := 0;
        hasApplyRcv := FALSE;

        // READ DATA FROM LINK TABLE 1
        tmpRcvApplyDetail.RESET;
        tmpRcvApplyDetail.SETFILTER("Whse. Receipt No.", '<>%1', "No.");
        tmpRcvApplyDetail.SETRANGE("Penerimaan No.", temporaryPenerimaan."No.");
        tmpRcvApplyDetail.SETRANGE("Penerimaan Line No.", temporaryPenerimaan."Line No.");
        tmpRcvApplyDetail.SETRANGE("Item No.", "Item No.");
        IF tmpRcvApplyDetail.FIND('-') THEN
        BEGIN
          REPEAT
            //temporaryPenerimaan.VALIDATE("Qty. On Apply",
              //temporaryPenerimaan."Qty. On Apply" - tmpRcvApplyDetail."Qty. to Apply");
            temporaryPenerimaan."Qty. On Apply" += tmpRcvApplyDetail."Qty. to Apply";
            temporaryPenerimaan.VALIDATE("Qty. On Apply");
          UNTIL tmpRcvApplyDetail.NEXT = 0;
          hasApplyRcv := TRUE;
        END;

        // READ DATA FROM LINK TABLE 2
        tmpRcvApplyDetail.RESET;
        tmpRcvApplyDetail.SETRANGE("Whse. Receipt No.", "No.");
        tmpRcvApplyDetail.SETRANGE("Whse. Receipt Line No.", "Line No.");
        tmpRcvApplyDetail.SETRANGE("Penerimaan No.", temporaryPenerimaan."No.");
        tmpRcvApplyDetail.SETRANGE("Penerimaan Line No.", temporaryPenerimaan."Line No.");
        tmpRcvApplyDetail.SETRANGE("Item No.", "Item No.");
        IF tmpRcvApplyDetail.FIND('-') THEN
        BEGIN
          //temporaryPenerimaan.VALIDATE("Qty. On Apply",
            //temporaryPenerimaan."Qty. On Apply" - tmpRcvApplyDetail."Qty. to Apply");
          tmpRcvApplyDetail2.RESET;
          tmpRcvApplyDetail2.SETRANGE("Whse. Receipt No.", "No.");
          tmpRcvApplyDetail2.SETFILTER("Whse. Receipt Line No.", '<>%1', "Line No.");
          tmpRcvApplyDetail2.SETFILTER("Penerimaan No.", temporaryPenerimaan."No.");
          tmpRcvApplyDetail2.SETRANGE("Penerimaan Line No.", temporaryPenerimaan."Line No.");
          tmpRcvApplyDetail2.SETFILTER("Item No.", "Item No.");
          IF tmpRcvApplyDetail2.FIND('-') THEN
          BEGIN
            REPEAT
              IF NOT hasApplyRcv THEN
                temporaryPenerimaan."Qty. Remaining" := temporaryPenerimaan.Quantity;
              hasApplyRcv := TRUE;

              IF tmpRcvApplyDetail2."Qty. to Apply" > 0 THEN
                temporaryPenerimaan."Qty. Remaining" -= tmpRcvApplyDetail2."Qty. to Apply";

              IF tmpRcvApplyDetail2."Qty. Applied" > 0 THEN
                temporaryPenerimaan."Qty. Remaining" -= tmpRcvApplyDetail2."Qty. Applied";
            UNTIL tmpRcvApplyDetail2.NEXT = 0;
          END;
          IF NOT hasApplyRcv THEN
            temporaryPenerimaan."Qty. Remaining" := temporaryPenerimaan.Quantity - temporaryPenerimaan."Qty. Applied";
          hasApplyRcv := TRUE;
          temporaryPenerimaan.VALIDATE("Qty. To Apply", tmpRcvApplyDetail."Qty. to Apply");
          //temporaryPenerimaan.VALIDATE("Qty. On Apply",tmpRcvApplyDetail."Qty. to Apply");
          //temporaryPenerimaan."Qty. To Apply" := tmpRcvApplyDetail."Qty. to Apply";
          temporaryPenerimaan."Qty. On Apply":= tmpRcvApplyDetail."Qty. to Apply";
        END
        ELSE
        BEGIN
          tmpRcvApplyDetail2.RESET;
          tmpRcvApplyDetail2.SETFILTER("Whse. Receipt No.", "No.");
          tmpRcvApplyDetail2.SETFILTER("Penerimaan No.", temporaryPenerimaan."No.");
          tmpRcvApplyDetail2.SETRANGE("Penerimaan Line No.", temporaryPenerimaan."Line No.");
          tmpRcvApplyDetail2.SETFILTER("Item No.", "Item No.");
          IF tmpRcvApplyDetail2.FIND('-') THEN
          BEGIN
            REPEAT
              IF NOT hasApplyRcv THEN
                temporaryPenerimaan."Qty. Remaining" := temporaryPenerimaan.Quantity;
              hasApplyRcv := TRUE;

              IF tmpRcvApplyDetail2."Qty. to Apply" > 0 THEN
                temporaryPenerimaan."Qty. Remaining" -= tmpRcvApplyDetail2."Qty. to Apply";

              IF tmpRcvApplyDetail2."Qty. Applied" > 0 THEN
                temporaryPenerimaan."Qty. Remaining" -= tmpRcvApplyDetail2."Qty. Applied";
            UNTIL tmpRcvApplyDetail2.NEXT = 0;
          END;
        END;

        // INSERT DATA TO TEMPORARY RECORD
        temporaryPenerimaan.INSERT;
      UNTIL tmpPenerimaanLine.NEXT = 0;

      // SHOW FORM
      CLEAR(PgPenerimaanApplied);
      PgPenerimaanApplied.SetDataSource(temporaryPenerimaan);
      PgPenerimaanApplied.SetSourceWarehouseReceipt(Rec);
      PgPenerimaanApplied.RUNMODAL;
    END;
