    PROCEDURE SetFilterUserID@1000000002(PLevelUserID@1000000000 : Code[50];PRespCtrUserID@1000000001 : Code[10]) RESULT : Text[1000];
    VAR
      LUserSetup@1000000002 : Record 91;
    BEGIN
      RESULT := '';
      IF PRespCtrUserID<>'' THEN
      BEGIN
         LUserSetup.RESET;
         LUserSetup.SETFILTER("P2B User Level",PLevelUserID);
         LUserSetup.SETFILTER("P2B Resp. Ctr. Filter",PRespCtrUserID);
         IF LUserSetup.FINDFIRST THEN
            REPEAT
               IF RESULT = '' THEN
                  RESULT := LUserSetup."User ID"
               ELSE
                  RESULT := RESULT + '|' + LUserSetup."User ID";
            UNTIL LUserSetup.NEXT = 0;
      END;
    END;

    PROCEDURE SetOptionPostingTo@1000000000();
    BEGIN
      IF "Posting To" = "Posting To"::"Item Journal" THEN BEGIN
        gBol_FromLocEditable := FALSE;
        gBol_ToLocEditable := FALSE;
        gBol_FromLocOwnerEditable := FALSE;
        gBol_ToLocOwnerEditable := FALSE;
        gBol_TransitLocEditable := FALSE;
        gBol_ShipToCodeEditable := FALSE;
        gBol_SalespersonEditable := TRUE;
        gBol_LoanTypeEditable := FALSE;
        gBol_LoanToCustEditable := FALSE;
        gBol_LoanDueDateEditable := FALSE;
      END ELSE BEGIN
        gBol_FromLocEditable := TRUE;
        gBol_ToLocEditable := TRUE;
        gBol_FromLocOwnerEditable := TRUE;
        gBol_ToLocOwnerEditable := TRUE;
        gBol_TransitLocEditable := TRUE;
        IF "Posting To" = "Posting To"::Loan THEN BEGIN
          gBol_ShipToCodeEditable := TRUE;
          gBol_SalespersonEditable := TRUE;
          gBol_LoanTypeEditable := FALSE;
          gBol_LoanToCustEditable := TRUE;
          gBol_LoanDueDateEditable := TRUE;
        END ELSE BEGIN
          gBol_ShipToCodeEditable := FALSE;
          gBol_SalespersonEditable := FALSE;
          gBol_LoanTypeEditable := FALSE;
          gBol_LoanToCustEditable := FALSE;
          gBol_LoanDueDateEditable := FALSE;
         END;
      END;
    END;

    PROCEDURE SetGenBusEnabled@1000000006();
    BEGIN
      IF "Posting To" <> 0 THEN
        gBol_GenBusEditable := FALSE
      ELSE BEGIN
        IF ("Document Type"= 'HRO40-PROMOSI') OR ("Document Type"='HRO40-PROMOSI LAIN') OR ("Document Type"='HRO40-LAIN') THEN BEGIN
          //IF (RespCenter."P2B Back Office") AND ("Gen. Bus. Posting Group"='RELASI MGT') THEN
          IF (RespCenter."P2B Job Title" = RespCenter."P2B Job Title"::"Back Office") AND ("Gen. Bus. Posting Group"='RELASI MGT') THEN
            gBol_GenBusEditable := FALSE
          ELSE
            gBol_GenBusEditable := TRUE;
        END ELSE IF ("Document Type"= 'HRO40-ATK-DIPAKAIHBS') OR ("Document Type"= 'HRO40-INV-IT') OR
          ("Document Type"= 'HRO40-INV-NON-IT') OR ("Document Type"='HRO40-INV-PEM')THEN
          gBol_GenBusEditable := FALSE
        ELSE
          gBol_GenBusEditable := TRUE
      END;

      //gBol_GenBusEditable := TRUE;
    END;
