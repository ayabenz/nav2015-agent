    LOCAL PROCEDURE InsertItemTrackingLine@2();
    VAR
      tempRE@1000 : TEMPORARY Record 337;
      n@1005 : Integer;
      tRE@1004 : Record 337;
      tESum@1003 : TEMPORARY Record 338;
      tSisaQty@1002 : Decimal;
      tQtyDigunakan@1001 : Decimal;
    BEGIN
      //T00.02 - START
      IF NOT FINDFIRST THEN EXIT;
      tempRE.RESET;
      tempRE.DELETEALL;

      //Insert into temporary table
      n := 0;
      tempRE.RESET;
      FINDFIRST;
      REPEAT
        TESTFIELD("Item No.");
        TESTFIELD("Location Code");

        //Negative Adjusment
        IF ("Entry Type" = "Entry Type"::"Negative Adjmt.") AND ("Line No." <> 0) THEN BEGIN
          tRE.RESET;
          tRE.SETRANGE("Source Type", 83);
          tRE.SETRANGE("Source Subtype", tRE."Source Subtype"::"3");
          tRE.SETRANGE("Source ID", "Journal Template Name");
          tRE.SETRANGE("Source Batch Name", "Journal Batch Name");
          tRE.SETRANGE("Source Ref. No.", "Line No.");
          //IF tRE.FINDFIRST THEN
            //tRE.DELETE;
          IF NOT tRE.FINDFIRST THEN BEGIN

            GetSaldoItemTracking("Item No.", "Location Code", tESum);

            tSisaQty := "Quantity (Base)";
            IF tESum.FINDFIRST THEN
            REPEAT
              tQtyDigunakan := 0;
              IF (tSisaQty > tESum."Total Quantity") THEN
                tQtyDigunakan := tESum."Total Quantity"
              ELSE
                tQtyDigunakan := tSisaQty;

              tSisaQty := tSisaQty - tQtyDigunakan;

              n += 1;
              tempRE.INIT;
              tempRE."Entry No." := n;
              tempRE."Lot No." := tESum."Lot No.";
              tempRE.Positive := FALSE;
              tempRE."Item No." := "Item No.";
              tempRE."Location Code" := "Location Code";
              tempRE.VALIDATE("Quantity (Base)", -tQtyDigunakan);
              tempRE."Reservation Status" := tempRE."Reservation Status"::Prospect;
              tempRE.Description := Description;
              tempRE."Creation Date" := TODAY;
              tempRE."Transferred from Entry No." := 0;
              tempRE."Source Type" := 83; //83. Table Item Journal Line
              tempRE."Source Subtype" := tempRE."Source Subtype"::"3"; //3. Entry Type = Neg. Adjm.
              tempRE."Source ID" := "Journal Template Name";
              tempRE."Source Batch Name" := "Journal Batch Name";
              tempRE."Source Prod. Order Line" := 0;
              tempRE."Source Ref. No." := "Line No.";
              tempRE."Appl.-to Item Entry" := 0;
              tempRE."Expected Receipt Date" := 0D;
              tempRE."Shipment Date" := TODAY;
              tempRE."Created By" := USERID;
              tempRE."Changed By" := '';
              tempRE."Qty. per Unit of Measure" := "Qty. per Unit of Measure";
              tempRE."Suppressed Action Msg." := FALSE;
              tempRE."Planning Flexibility" := tempRE."Planning Flexibility"::Unlimited;
              tempRE."Expiration Date" := tESum."Expiration Date";
              tempRE.Correction := FALSE;
              tempRE."Item Tracking" := tempRE."Item Tracking"::"Lot No.";
              tempRE.INSERT;
            UNTIL (tESum.NEXT = 0) OR (tSisaQty = 0);
          END;
        END;

        //Positive Adjusment
        IF ("Entry Type" = "Entry Type"::"Positive Adjmt.") AND ("Line No." <> 0) THEN BEGIN

          tRE.RESET;
          tRE.SETRANGE("Source Type", 83);
          tRE.SETRANGE("Source Subtype", tRE."Source Subtype"::"2");
          tRE.SETRANGE("Source ID", "Journal Template Name");
          tRE.SETRANGE("Source Batch Name", "Journal Batch Name");
          tRE.SETRANGE("Source Ref. No.", "Line No.");
          //IF tRE.FINDFIRST THEN
            //tRE.DELETE;

          IF NOT tRE.FINDFIRST THEN BEGIN
            n += 1;
            tempRE.INIT;
            tempRE."Entry No." := n;
            tempRE."Lot No." := FORMAT(TODAY, 0, '<year4>/<month,2>/<day,2>');
            tempRE.Positive := TRUE;
            tempRE."Item No." := "Item No.";
            tempRE."Location Code" := "Location Code";
            tempRE.VALIDATE("Quantity (Base)", "Quantity (Base)");
            tempRE."Reservation Status" := tempRE."Reservation Status"::Prospect;
            tempRE.Description := Description;
            tempRE."Creation Date" := TODAY;
            tempRE."Transferred from Entry No." := 0;
            tempRE."Source Type" := 83; //83. Table Item Journal Line
            tempRE."Source Subtype" := tempRE."Source Subtype"::"2"; //2. Entry Type = Pos. Adjm.
            tempRE."Source ID" := "Journal Template Name";
            tempRE."Source Batch Name" := "Journal Batch Name";
            tempRE."Source Prod. Order Line" := 0;
            tempRE."Source Ref. No." := "Line No.";
            tempRE."Appl.-to Item Entry" := 0;
            tempRE."Expected Receipt Date" := TODAY;
            tempRE."Shipment Date" := 0D;
            tempRE."Created By" := USERID;
            tempRE."Changed By" := '';
            tempRE."Qty. per Unit of Measure" := "Qty. per Unit of Measure";
            tempRE."Suppressed Action Msg." := FALSE;
            tempRE."Planning Flexibility" := tempRE."Planning Flexibility"::Unlimited;
            tempRE."Expiration Date" := CALCDATE('+3Y', TODAY);
            tempRE.Correction := FALSE;
            tempRE."Item Tracking" := tempRE."Item Tracking"::"Lot No.";
            tempRE.INSERT;
          END;
        END;
      UNTIL NEXT = 0;

      tempRE.RESET;
      IF NOT tempRE.FINDFIRST THEN EXIT;

      //Insert into Reservation Entry Table
      n := 0;
      tRE.LOCKTABLE;
      tRE.RESET;
      IF tRE.FINDLAST THEN
        n := tRE."Entry No.";

      REPEAT
        n += 1;
        tRE.INIT;
        tRE := tempRE;
        tRE."Entry No." := n;
        tRE.INSERT;
      UNTIL tempRE.NEXT = 0;

      MESSAGE('Completed');

      //T00.02 - END
    END;

    PROCEDURE GetSaldoItemTracking@1000000003(ItemNo@1000000000 : Code[20];LocationCode@1000000001 : Code[10];VAR tES@1000000002 : TEMPORARY Record 338);
    VAR
      tILE@1000000003 : Record 32;
      tRE@1000000004 : Record 337;
      n@1000000005 : Integer;
    BEGIN
      //T00.02 - START
      tES.RESET;
      tES.DELETEALL;
      n := 0;

      tILE.RESET;
      tILE.SETRANGE("Item No.", ItemNo);
      tILE.SETRANGE("Location Code", LocationCode);
      tILE.SETRANGE(Open, TRUE);
      IF tILE.FINDFIRST THEN
      REPEAT
        tES.RESET;
        tES.SETRANGE("Lot No.", tILE."Lot No.");
        tES.SETRANGE("Expiration Date", tILE."Expiration Date");
        IF tES.FINDFIRST THEN
        BEGIN
          tES."Total Quantity" += tILE."Remaining Quantity";
          tES.MODIFY;
        END
        ELSE
        BEGIN
          n += 1;
          tES.INIT;
          tES."Entry No." := n;
          tES."Lot No." := tILE."Lot No.";
          tES."Expiration Date" := tILE."Expiration Date";
          tES."Total Quantity" := tILE."Remaining Quantity";
          tES.INSERT;
        END;
      UNTIL tILE.NEXT = 0;

      tES.RESET;
      IF tES.FINDFIRST THEN
      REPEAT
        tRE.RESET;
        tRE.SETRANGE(Positive, FALSE);
        tRE.SETRANGE("Item No.", ItemNo);
        tRE.SETRANGE("Location Code", LocationCode);
        tRE.SETRANGE("Lot No.", tES."Lot No.");
        tRE.SETRANGE("Expiration Date", tES."Expiration Date");
        IF tRE.FINDFIRST THEN
        BEGIN
          REPEAT
            tES."Total Reserved Quantity" += -tRE."Quantity (Base)";
          UNTIL tRE.NEXT = 0;
          tES."Total Available Quantity" := tES."Total Quantity" - tES."Total Reserved Quantity";
          tES.MODIFY;
        END;
      UNTIL tES.NEXT = 0;

      tES.RESET;
      tES.SETFILTER("Total Quantity", '<=0');
      IF tES.FIND('-') THEN
        tES.DELETEALL;

      tES.RESET;
      //T00.02 - END
    END;
